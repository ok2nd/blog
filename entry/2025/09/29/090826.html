Claude）Python）シンプルなWindows用ランチャーアプリ作成。

2025-09-29 · コンピュータ ＜AI＞

Pythonで作成したWindows用アプリが増えてきた。

これらを起動するランチャーアプリが欲しいと思って、作成した。

設定できる項目を少なくして、極力シンプルなものにした。

設定できるのは、ボタンサイズと横に並べるボタンの数だけ。

実行するアプリは、テキスト形式で「アプリ名：アプリのWindowsフォルダーパス」で入力。

Pythonで作成した以外の一般的なWindowsアプリも登録は可能。

【Pythonプログラムコード】

※ 2026.1.8改訂。

import tkinter as tk
from tkinter import ttk, messagebox, scrolledtext
import subprocess
import os
import json

class WindowsLauncher:
    def __init__(self, root):
        self.root = root
        self.root.title("Windows ランチャー")

        # 設定ファイル
        self.config_file = "launcher_config.json"

        # デフォルト設定
        self.button_size = 100
        self.buttons_per_row = 5
        self.apps = []
        self.window_x = 100
        self.window_y = 100

        # 設定読込
        self.load_config()

        # UI 作成
        self.create_widgets()

        # 起動時のみ：サイズ＋位置を設定
        self.set_window_geometry_startup()

        # 閉じる処理
        self.root.protocol("WM_DELETE_WINDOW", self.on_closing)

        # ボタン生成
        self.update_app_buttons()

    # ==================================================
    # ウィンドウ制御（★ここが重要）
    # ==================================================
    def set_window_geometry_startup(self):
        """起動時のみ：サイズ＋位置を指定"""
        w, h = self.calculate_window_size()
        self.root.geometry(f"{w}x{h}+{self.window_x}+{self.window_y}")

    def update_window_size_only(self):
        """サイズだけ変更（位置は触らない）"""
        w, h = self.calculate_window_size()
        self.root.geometry(f"{w}x{h}")

    def calculate_window_size(self):
        """現在の設定からウィンドウサイズを計算"""
        button_pixel_width = max(self.button_size // 8 * 8, 80)
        padding_per_button = 10
        content_width = (button_pixel_width + padding_per_button) * self.buttons_per_row
        ui_base_width = 100
        window_width = max(content_width + ui_base_width, 400)
        window_height = 600
        return window_width, window_height

    def on_closing(self):
        """終了時に位置保存"""
        try:
            self.window_x = self.root.winfo_x()
            self.window_y = self.root.winfo_y()
        except Exception:
            pass
        self.save_current_config()
        self.root.destroy()

    # ==================================================
    # 設定保存・読込
    # ==================================================
    def save_current_config(self):
        config = {
            "button_size": self.button_size,
            "buttons_per_row": self.buttons_per_row,
            "apps": self.apps,
            "window_x": self.window_x,
            "window_y": self.window_y,
        }
        try:
            with open(self.config_file, "w", encoding="utf-8") as f:
                json.dump(config, f, ensure_ascii=False, indent=2)
        except Exception:
            pass

    def load_config(self):
        if not os.path.exists(self.config_file):
            return
        try:
            with open(self.config_file, "r", encoding="utf-8") as f:
                config = json.load(f)
            self.button_size = config.get("button_size", 100)
            self.buttons_per_row = config.get("buttons_per_row", 5)
            self.apps = config.get("apps", [])
            self.window_x = config.get("window_x", 100)
            self.window_y = config.get("window_y", 100)
        except Exception as e:
            messagebox.showerror("エラー", f"設定読み込みエラー: {e}")

    # ==================================================
    # UI
    # ==================================================
    def create_widgets(self):
        main = ttk.Frame(self.root, padding=10)
        main.pack(fill="both", expand=True)

        notebook = ttk.Notebook(main)
        notebook.pack(fill="both", expand=True)

        self.launcher_frame = ttk.Frame(notebook)
        settings_frame = ttk.Frame(notebook)

        notebook.add(self.launcher_frame, text="ランチャー")
        notebook.add(settings_frame, text="設定")

        self.create_settings_tab(settings_frame)

        canvas = tk.Canvas(self.launcher_frame)
        scrollbar = ttk.Scrollbar(self.launcher_frame, orient="vertical", command=canvas.yview)
        self.scrollable_frame = ttk.Frame(canvas)

        self.scrollable_frame.bind(
            "<Configure>",
            lambda e: canvas.configure(scrollregion=canvas.bbox("all"))
        )

        canvas.create_window((0, 0), window=self.scrollable_frame, anchor="nw")
        canvas.configure(yscrollcommand=scrollbar.set)

        canvas.pack(side="left", fill="both", expand=True)
        scrollbar.pack(side="right", fill="y")

    # ==================================================
    # 設定タブ
    # ==================================================
    def create_settings_tab(self, parent):
        frame = ttk.Frame(parent, padding=10)
        frame.pack(fill="both", expand=True)

        ttk.Label(frame, text="ボタンサイズ").pack(anchor="w")
        self.size_var = tk.StringVar(value=str(self.button_size))
        ttk.Entry(frame, textvariable=self.size_var).pack(anchor="w")

        ttk.Label(frame, text="横に並べる数").pack(anchor="w", pady=(10, 0))
        self.row_var = tk.StringVar(value=str(self.buttons_per_row))
        ttk.Entry(frame, textvariable=self.row_var).pack(anchor="w")

        ttk.Label(frame, text="アプリ一覧（名前:パス）").pack(anchor="w", pady=(20, 5))
        self.apps_text = scrolledtext.ScrolledText(frame, height=15)
        self.apps_text.pack(fill="both", expand=True)

        self.update_apps_text()

        ttk.Button(frame, text="設定を適用", command=self.apply_settings).pack(pady=10)

    def update_apps_text(self):
        self.apps_text.delete(1.0, tk.END)
        for app in self.apps:
            self.apps_text.insert(tk.END, f"{app['name']}:{app['path']}\n")

    def parse_apps_text(self):
        apps = []
        for line in self.apps_text.get(1.0, tk.END).splitlines():
            if ":" in line:
                name, path = line.split(":", 1)
                apps.append({"name": name.strip(), "path": path.strip()})
        return apps

    def apply_settings(self):
        try:
            self.button_size = int(self.size_var.get())
            self.buttons_per_row = int(self.row_var.get())
            self.apps = self.parse_apps_text()

            # ★位置は変えない（サイズのみ）
            self.update_window_size_only()
            self.update_app_buttons()

        except ValueError:
            messagebox.showerror("エラー", "数値が正しくありません")

    # ==================================================
    # ランチャー
    # ==================================================
    def update_app_buttons(self):
        for w in self.scrollable_frame.winfo_children():
            w.destroy()

        for i, app in enumerate(self.apps):
            r = i // self.buttons_per_row
            c = i % self.buttons_per_row

            btn = tk.Button(
                self.scrollable_frame,
                text=app["name"],
                width=self.button_size // 8,
                height=self.button_size // 20,
                wraplength=self.button_size - 10,
                justify="center",
                font=("Arial", 9),
                command=lambda p=app["path"]: subprocess.Popen(p, shell=True)
            )
            btn.grid(row=r, column=c, padx=5, pady=5, sticky="nsew")

def main():
    root = tk.Tk()
    WindowsLauncher(root)
    root.mainloop()

if __name__ == "__main__":
    main()
