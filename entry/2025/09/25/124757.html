Claude）Python）ローカル）写真のシャープネス、コントラスト、明度、彩度を調整するWindowsアプリ作成。野鳥写真で、等倍高画質化をTopaz Labs、Aiarty Image Enhancerと比較。

2025-09-25 · 写真・カメラ ＜AI＞ 野鳥

Pythonで「写真のシャープネス、コントラスト、明度、彩度を調整する」Windowsアプリを作成。「Claude」で作成。

【機能概略】

  • 写真画像(JPEG,PNG)をウインドウに、ドラッグ＆ドロップで取り込み。
  • 写真画像は、縮小サイズと等倍サイズで確認できる。等倍サイズでは、マウスドラッグで表示位置を変更できる。
  • シャープネス、コントラスト、明度、彩度を調整できる。
  • 調整後の画像も合わせて確認できる。
  • 調整結果はダウンロードできる。

【Pythonプログラム】

作成されたプログラムを、「画像調整アプリ.pyw」（名前は任意）という名前にして、デスクトップに置いておき、ダブルクリックで実行する。

【Pythonライブラリインストール】

次のライブラリが必要なので、インストールする。

pip install pillow tkinterdnd2

【Pythonプログラムコード】

import tkinter as tk
from tkinter import ttk, filedialog, messagebox
import tkinter.dnd as dnd
from tkinterdnd2 import TkinterDnD, DND_FILES
from PIL import Image, ImageTk, ImageEnhance
import os
import math

class ImageAdjustmentApp:
    def __init__(self, root):
        self.root = root
        self.root.title("画像調整アプリ")
        self.root.geometry("1200x800")
        
        # 変数の初期化
        self.original_image = None
        self.current_image = None
        self.display_image = None
        self.adjusted_image = None
        self.photo_original = None
        self.photo_adjusted = None
        self.original_filename = None  # オリジナルファイル名を保存
        self.original_dir = None  # オリジナル画像のフォルダーを保存
        self.last_save_dir = None  # 前回の保存フォルダーを記憶
        
        # 調整パラメータ
        self.sharpness = tk.DoubleVar(value=1.0)
        self.contrast = tk.DoubleVar(value=1.0)
        self.brightness = tk.DoubleVar(value=1.0)
        self.saturation = tk.DoubleVar(value=1.0)
        
        # 表示設定
        self.display_width = 400
        self.is_full_size = False
        self.pan_x = 0
        self.pan_y = 0
        self.last_mouse_x = 0
        self.last_mouse_y = 0
        
        self.setup_ui()
        self.root.drop_target_register(DND_FILES)
        self.root.dnd_bind('<<Drop>>', self.drop_file)
        
        # ウィンドウサイズ変更時のイベントバインド
        self.root.bind('<Configure>', self.on_window_resize)
        
    def setup_ui(self):
        # メインフレーム
        main_frame = ttk.Frame(self.root, padding="10")
        main_frame.grid(row=0, column=0, sticky=(tk.W, tk.E, tk.N, tk.S))
        
        # 左側：画像表示エリア
        image_frame = ttk.Frame(main_frame)
        image_frame.grid(row=0, column=0, padx=(0, 10), sticky=(tk.W, tk.E, tk.N, tk.S))
        
        # 画像表示ラベル（オリジナル）
        self.original_label = ttk.Label(image_frame, text="画像をドラッグ&ドロップしてください", 
                                       width=50, background="lightgray", anchor="center")
        self.original_label.grid(row=0, column=0, padx=5, pady=5)
        self.original_label.bind("<Button-1>", self.select_image)
        self.original_label.bind("<B1-Motion>", self.pan_image)
        self.original_label.bind("<ButtonPress-1>", self.start_pan)
        
        # 画像表示ラベル（調整後）
        self.adjusted_label = ttk.Label(image_frame, text="調整後の画像", 
                                       width=50, background="lightgray", anchor="center")
        self.adjusted_label.grid(row=0, column=1, padx=5, pady=5)
        self.adjusted_label.bind("<B1-Motion>", self.pan_image)
        self.adjusted_label.bind("<ButtonPress-1>", self.start_pan)
        
        # 表示切り替えボタン
        button_frame = ttk.Frame(image_frame)
        button_frame.grid(row=1, column=0, columnspan=2, pady=5)
        
        ttk.Button(button_frame, text="縮小サイズ", command=self.set_scaled_view).pack(side=tk.LEFT, padx=5)
        ttk.Button(button_frame, text="等倍サイズ", command=self.set_full_size_view).pack(side=tk.LEFT, padx=5)
        
        # 右側：調整パネル
        control_frame = ttk.Frame(main_frame)
        control_frame.grid(row=0, column=1, sticky=(tk.W, tk.E, tk.N, tk.S))
        
        # 調整コントロール
        self.create_adjustment_control(control_frame, "シャープネス", self.sharpness, 0, 0.0, 5.0, 0.1, self.reset_sharpness)
        self.create_adjustment_control(control_frame, "コントラスト", self.contrast, 1, 0.8, 1.2, 0.02, self.reset_contrast)
        self.create_adjustment_control(control_frame, "明度", self.brightness, 2, 0.8, 1.2, 0.02, self.reset_brightness)
        self.create_adjustment_control(control_frame, "彩度", self.saturation, 3, 0.8, 1.2, 0.02, self.reset_saturation)
        
        # 全体リセットボタン
        ttk.Button(control_frame, text="全てリセット", command=self.reset_all).grid(row=4, column=0, columnspan=3, pady=10, sticky=(tk.W, tk.E))
        
        # ダウンロードボタン
        ttk.Button(control_frame, text="画像をダウンロード", command=self.download_image).grid(row=5, column=0, columnspan=3, pady=10, sticky=(tk.W, tk.E))
        
        # グリッドの重み設定
        self.root.columnconfigure(0, weight=1)
        self.root.rowconfigure(0, weight=1)
        main_frame.columnconfigure(0, weight=2)
        main_frame.columnconfigure(1, weight=1)
        main_frame.rowconfigure(0, weight=1)
        
    def create_adjustment_control(self, parent, label, variable, row, min_val, max_val, resolution, reset_func):
        # ラベル
        ttk.Label(parent, text=label).grid(row=row, column=0, sticky=tk.W, pady=5)
        
        # スライダー
        scale = ttk.Scale(parent, from_=min_val, to=max_val, variable=variable, 
                         orient=tk.HORIZONTAL, command=self.on_scale_change)
        scale.grid(row=row, column=1, sticky=(tk.W, tk.E), padx=5)
        
        # スライダーの解像度を設定
        if resolution < 0.1:
            # 0.02刻みの場合、スライダーの分解能を細かく設定
            scale.configure(length=300)
        
        # 値表示（小数点2桁）
        value_var = tk.StringVar()
        value_label = ttk.Label(parent, textvariable=value_var, width=6)
        value_label.grid(row=row, column=2, padx=5)
        
        # 値を適切な桁数で更新する関数を保存
        def update_display():
            if resolution < 0.1:
                value_var.set(f"{variable.get():.2f}")  # 0.02刻みの場合は小数点2桁
            else:
                value_var.set(f"{variable.get():.1f}")  # 0.1刻みの場合は小数点1桁
        
        variable.trace_add('write', lambda *args: update_display())
        update_display()  # 初期値を設定
        
        # リセットボタン
        ttk.Button(parent, text="リセット", command=reset_func).grid(row=row, column=3, padx=5)
        
        parent.columnconfigure(1, weight=1)

    def on_scale_change(self, *args):
        """スライダー変更時の処理（適切な桁数に丸める）"""
        # シャープネスは小数点1桁、その他は小数点2桁に丸める
        self.sharpness.set(round(self.sharpness.get(), 1))
        self.contrast.set(round(self.contrast.get() / 0.02) * 0.02)  # 0.02刻み
        self.brightness.set(round(self.brightness.get() / 0.02) * 0.02)  # 0.02刻み
        self.saturation.set(round(self.saturation.get() / 0.02) * 0.02)  # 0.02刻み
        self.update_image()
    
    def on_window_resize(self, event):
        """ウィンドウサイズ変更時の処理"""
        if event.widget == self.root and self.current_image:
            # ウィンドウの幅に基づいて表示サイズを調整
            window_width = self.root.winfo_width()
            # コントロールパネルの幅を考慮して画像表示領域を計算
            available_width = max(200, (window_width - 400) // 2)  # 最小200px
            self.display_width = min(600, available_width)  # 最大600px
            
            self.update_display()
            self.update_adjusted_display()

    def drop_file(self, event):
        files = event.data.split()
        if files:
            file_path = files[0].strip('{}')  # 波括弧を除去
            self.load_image(file_path)
    
    def select_image(self, event=None):
        file_path = filedialog.askopenfilename(
            title="画像を選択",
            filetypes=[("画像ファイル", "*.jpg *.jpeg *.png *.bmp *.gif")]
        )
        if file_path:
            self.load_image(file_path)
    
    def load_image(self, file_path):
        try:
            self.original_image = Image.open(file_path)
            self.current_image = self.original_image.copy()
            
            # オリジナルファイル名を保存（拡張子なし）
            self.original_filename = os.path.splitext(os.path.basename(file_path))[0]
            # オリジナル画像のフォルダーを保存
            self.original_dir = os.path.dirname(file_path)
            
            self.pan_x = 0
            self.pan_y = 0
            self.update_display()
            self.update_image()  # 現在の調整値を適用
        except Exception as e:
            messagebox.showerror("エラー", f"画像の読み込みに失敗しました: {str(e)}")
    
    def update_image(self, *args):
        if self.current_image is None:
            return
        
        # 画像調整を適用
        img = self.current_image.copy()
        
        # シャープネス
        enhancer = ImageEnhance.Sharpness(img)
        img = enhancer.enhance(self.sharpness.get())
        
        # コントラスト（0.8～1.2の範囲をそのまま使用）
        enhancer = ImageEnhance.Contrast(img)
        img = enhancer.enhance(self.contrast.get())
        
        # 明度（0.8～1.2の範囲をそのまま使用）
        enhancer = ImageEnhance.Brightness(img)
        img = enhancer.enhance(self.brightness.get())
        
        # 彩度（0.8～1.2の範囲をそのまま使用）
        enhancer = ImageEnhance.Color(img)
        img = enhancer.enhance(self.saturation.get())
        
        self.adjusted_image = img
        self.update_adjusted_display()
    
    def update_display(self):
        if self.current_image is None:
            return
        
        if self.is_full_size:
            # 等倍表示
            img_width, img_height = self.current_image.size
            
            # 表示領域のサイズ
            display_width = self.display_width
            display_height = int(display_width * 0.75)  # アスペクト比を調整
            
            # 切り取り領域の計算
            left = max(0, self.pan_x)
            top = max(0, self.pan_y)
            right = min(img_width, left + display_width)
            bottom = min(img_height, top + display_height)
            
            display_img = self.current_image.crop((left, top, right, bottom))
        else:
            # 縮小表示
            display_img = self.current_image.copy()
            display_img.thumbnail((self.display_width, self.display_width), Image.Resampling.LANCZOS)
        
        self.photo_original = ImageTk.PhotoImage(display_img)
        self.original_label.config(image=self.photo_original, text="")
    
    def update_adjusted_display(self):
        if self.adjusted_image is None:
            return
        
        if self.is_full_size:
            # 等倍表示
            img_width, img_height = self.adjusted_image.size
            
            # 表示領域のサイズ
            display_width = self.display_width
            display_height = int(display_width * 0.75)
            
            # 切り取り領域の計算
            left = max(0, self.pan_x)
            top = max(0, self.pan_y)
            right = min(img_width, left + display_width)
            bottom = min(img_height, top + display_height)
            
            display_img = self.adjusted_image.crop((left, top, right, bottom))
        else:
            # 縮小表示
            display_img = self.adjusted_image.copy()
            display_img.thumbnail((self.display_width, self.display_width), Image.Resampling.LANCZOS)
        
        self.photo_adjusted = ImageTk.PhotoImage(display_img)
        self.adjusted_label.config(image=self.photo_adjusted, text="")
    
    def set_scaled_view(self):
        self.is_full_size = False
        self.update_display()
        self.update_adjusted_display()
    
    def set_full_size_view(self):
        self.is_full_size = True
        self.update_display()
        self.update_adjusted_display()
    
    def start_pan(self, event):
        self.last_mouse_x = event.x
        self.last_mouse_y = event.y
    
    def pan_image(self, event):
        if self.current_image is None or not self.is_full_size:
            return
        
        dx = event.x - self.last_mouse_x
        dy = event.y - self.last_mouse_y
        
        self.pan_x -= dx
        self.pan_y -= dy
        
        # 境界チェック
        img_width, img_height = self.current_image.size
        max_pan_x = max(0, img_width - self.display_width)
        max_pan_y = max(0, img_height - int(self.display_width * 0.75))
        
        self.pan_x = max(0, min(max_pan_x, self.pan_x))
        self.pan_y = max(0, min(max_pan_y, self.pan_y))
        
        self.last_mouse_x = event.x
        self.last_mouse_y = event.y
        
        self.update_display()
        self.update_adjusted_display()
    
    def reset_sharpness(self):
        self.sharpness.set(1.0)
        self.update_image()
    
    def reset_contrast(self):
        self.contrast.set(1.0)
        self.update_image()
    
    def reset_brightness(self):
        self.brightness.set(1.0)
        self.update_image()
    
    def reset_saturation(self):
        self.saturation.set(1.0)
        self.update_image()
    
    def reset_all(self):
        self.sharpness.set(1.0)
        self.contrast.set(1.0)
        self.brightness.set(1.0)
        self.saturation.set(1.0)
        self.update_image()
    
    def download_image(self):
        if self.adjusted_image is None:
            messagebox.showwarning("警告", "調整された画像がありません")
            return
        
        try:
            # ファイル名の生成（オリジナルファイル名 + 調整値）
            s_val = f"{self.sharpness.get():.1f}"
            c_val = f"{self.contrast.get():.2f}"
            b_val = f"{self.brightness.get():.2f}"
            sat_val = f"{self.saturation.get():.2f}"
            
            # オリジナルファイル名がない場合のデフォルト名
            base_name = self.original_filename if self.original_filename else "adjusted"
            
            filename = f"{base_name}_S{s_val}-C{c_val}-B{b_val}-S{sat_val}.jpg"
            
            # 保存先フォルダーを決定（優先順位：オリジナル画像→前回保存→デスクトップ）
            if self.original_dir and os.path.exists(self.original_dir):
                initial_dir = self.original_dir
            elif self.last_save_dir and os.path.exists(self.last_save_dir):
                initial_dir = self.last_save_dir
            else:
                # デスクトップフォルダーを取得
                try:
                    initial_dir = os.path.join(os.path.join(os.environ['USERPROFILE']), 'Desktop')
                    if not os.path.exists(initial_dir):
                        initial_dir = os.path.expanduser("~")
                except:
                    initial_dir = os.path.expanduser("~")
            
            # 保存ダイアログを表示（フォルダー変更可能）
            file_path = filedialog.asksaveasfilename(
                title="調整された画像を保存",
                defaultextension=".jpg",
                initialfile=filename,
                filetypes=[
                    ("JPEG画像", "*.jpg"),
                    ("PNG画像", "*.png"),
                    ("すべてのファイル", "*.*")
                ],
                initialdir=initial_dir
            )
            
            if file_path:
                # 保存フォルダーを記憶
                self.last_save_dir = os.path.dirname(file_path)
                
                # 調整された画像を保存
                save_image = self.adjusted_image.copy()
                
                # ファイル拡張子に応じて保存形式を決定
                if file_path.lower().endswith(('.jpg', '.jpeg')):
                    # JPEG保存の場合はRGBに変換
                    if save_image.mode != 'RGB':
                        save_image = save_image.convert('RGB')
                    save_image.save(file_path, 'JPEG', quality=95)
                elif file_path.lower().endswith('.png'):
                    # PNG保存
                    save_image.save(file_path, 'PNG')
                else:
                    # その他の場合はJPEGとして保存
                    if save_image.mode != 'RGB':
                        save_image = save_image.convert('RGB')
                    save_image.save(file_path + '.jpg', 'JPEG', quality=95)
                    file_path += '.jpg'
                
                # messagebox.showinfo("保存完了", f"画像を保存しました:\n{os.path.basename(file_path)}\n\n保存先:\n{os.path.dirname(file_path)}")
            else:
                messagebox.showinfo("キャンセル", "保存がキャンセルされました")
                
        except Exception as e:
            messagebox.showerror("保存エラー", f"画像の保存に失敗しました:\n{str(e)}\n\n別の場所に保存してみてください。")

def main():
    root = TkinterDnD.Tk()
    app = ImageAdjustmentApp(root)
    root.mainloop()

if __name__ == "__main__":
    main()
