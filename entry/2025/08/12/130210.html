Claude）Python）「写真をEXIF機種毎のフォルダーに分類」Windowsアプリ作成。

2025-08-12 · 写真・カメラ ＜AI＞ コンピュータ

Pythonで「写真をEXIF機種毎のフォルダーに分類」Windowsアプリを作成。

生成AI「Claude」「Gemini」「Copilot」で、以下のプロンプトで作成した。

【生成AI プロンプト】

Windows環境のPythonでアプリを作って。

Windowsフォルダ内のJPEG画像ファイルをEXIFのカメラ機種毎のフォルダーに分類するアプリ。

  • Windowsフォルダ内のJPEG画像ファイルのEXIFの「カメラのモデル（Model）」を読み取って、モデル名のフォルダーを作成し、そこに「移動」する。（コピーではない。）
  • 処理対象フォルダーの階層を3階層下まで指定できる。何階層目を処理するかはラジオボタンで選択できる。
  • 2階層以降を指定した場合も、モデル名のフォルダーは、それぞれのフォルダーの下に作る。
  • 処理対象フォルダー（最上位階層）は、アプリのGUIで指定できるようにする。
  • 処理対象フォルダー（最上位階層）選択用のフォルダー入力欄は、Windowsフォルダー等からコピー&ペーストできる。
  • 処理対象フォルダー（最上位階層）選択ボタンを押したら、その時点のフォルダー入力欄のパスを開く。
  • 実行状況のログ表示は不要。完了時だけ、完了メッセージを出す。

出来が良かった「Claude」版を採用。

【Pythonプログラムコード】
import os
import tkinter as tk
from tkinter import ttk, filedialog, messagebox
import shutil
from PIL import Image
from PIL.ExifTags import TAGS
import threading

class JPEGClassifierApp:
    def __init__(self, root):
        self.root = root
        self.root.title("JPEG画像カメラ機種別分類ツール")
        self.root.geometry("600x300")
        
        # フォルダーパス入力フレーム
        folder_frame = ttk.Frame(root, padding="10")
        folder_frame.grid(row=0, column=0, sticky=(tk.W, tk.E))
        
        ttk.Label(folder_frame, text="処理対象フォルダー:").grid(row=0, column=0, sticky=tk.W, pady=5)
        
        # フォルダーパス入力欄
        self.folder_var = tk.StringVar()
        self.folder_entry = ttk.Entry(folder_frame, textvariable=self.folder_var, width=50)
        self.folder_entry.grid(row=1, column=0, sticky=(tk.W, tk.E), pady=5)
        
        # フォルダー選択ボタン
        ttk.Button(folder_frame, text="フォルダー選択", 
                  command=self.browse_folder).grid(row=1, column=1, padx=(10, 0), pady=5)
        
        # 処理階層選択フレーム
        level_frame = ttk.Frame(root, padding="10")
        level_frame.grid(row=1, column=0, sticky=(tk.W, tk.E))
        
        ttk.Label(level_frame, text="処理階層:").grid(row=0, column=0, sticky=tk.W, pady=5)
        
        self.level_var = tk.IntVar(value=1)
        
        # ラジオボタン
        ttk.Radiobutton(level_frame, text="1階層目のみ", variable=self.level_var, 
                       value=1).grid(row=1, column=0, sticky=tk.W)
        ttk.Radiobutton(level_frame, text="2階層目まで", variable=self.level_var, 
                       value=2).grid(row=1, column=1, sticky=tk.W, padx=20)
        ttk.Radiobutton(level_frame, text="3階層目まで", variable=self.level_var, 
                       value=3).grid(row=1, column=2, sticky=tk.W, padx=20)
        
        # 実行ボタンフレーム
        button_frame = ttk.Frame(root, padding="10")
        button_frame.grid(row=2, column=0, sticky=(tk.W, tk.E))
        
        self.execute_button = ttk.Button(button_frame, text="実行", 
                                        command=self.start_classification)
        self.execute_button.grid(row=0, column=0, pady=20)
        

        
        # グリッドの重み設定
        root.columnconfigure(0, weight=1)
        folder_frame.columnconfigure(0, weight=1)
        button_frame.columnconfigure(0, weight=1)
    
    def browse_folder(self):
        """フォルダー選択ダイアログを開く"""
        current_path = self.folder_var.get()
        if current_path and os.path.exists(current_path):
            # 現在のパスが存在する場合はそれを初期ディレクトリに
            initial_dir = current_path
        else:
            initial_dir = os.path.expanduser("~")
            
        folder_selected = filedialog.askdirectory(initialdir=initial_dir)
        if folder_selected:
            self.folder_var.set(folder_selected)
    
    def get_camera_model(self, image_path):
        """JPEG画像からカメラモデル情報を取得"""
        try:
            image = Image.open(image_path)
            exifdata = image.getexif()
            
            for tag_id in exifdata:
                tag = TAGS.get(tag_id, tag_id)
                if tag == "Model":
                    model = exifdata.get(tag_id)
                    if model:
                        # ファイル名に使用できない文字を除去
                        model = "".join(c for c in model if c.isalnum() or c in (' ', '-', '_')).strip()
                        return model
            return None
        except Exception:
            return None
    
    def collect_jpeg_files(self, base_path, max_level):
        """指定された階層までのJPEGファイルを収集（親ディレクトリ情報も含む）"""
        jpeg_files = []
        
        def scan_directory(path, current_level):
            if current_level > max_level:
                return
            
            try:
                for item in os.listdir(path):
                    item_path = os.path.join(path, item)
                    
                    if os.path.isfile(item_path):
                        # JPEGファイルかチェック
                        if item.lower().endswith(('.jpg', '.jpeg')):
                            # ファイルパスと親ディレクトリのペアで保存
                            jpeg_files.append((item_path, path))
                    elif os.path.isdir(item_path) and current_level < max_level:
                        # 次の階層をスキャン
                        scan_directory(item_path, current_level + 1)
            except PermissionError:
                pass  # アクセス権限がない場合はスキップ
        
        scan_directory(base_path, 1)
        return jpeg_files
    
    def classify_images(self):
        """画像の分類処理"""
        base_folder = self.folder_var.get()
        max_level = self.level_var.get()
        
        if not base_folder or not os.path.exists(base_folder):
            messagebox.showerror("エラー", "有効なフォルダーを指定してください。")
            return
        
        # JPEGファイルを収集（親ディレクトリ情報も含む）
        jpeg_files = self.collect_jpeg_files(base_folder, max_level)
        
        if not jpeg_files:
            messagebox.showinfo("完了", "処理対象のJPEGファイルが見つかりませんでした。")
            return
        
        processed_count = 0
        created_folders = set()
        
        for image_path, parent_dir in jpeg_files:
            model = self.get_camera_model(image_path)
            
            if model:
                # 親ディレクトリの下にカメラモデル用のフォルダーを作成
                model_folder = os.path.join(parent_dir, model)
                
                if model_folder not in created_folders:
                    if not os.path.exists(model_folder):
                        try:
                            os.makedirs(model_folder)
                        except Exception as e:
                            continue
                    created_folders.add(model_folder)
                
                # 画像ファイルを移動
                try:
                    filename = os.path.basename(image_path)
                    destination = os.path.join(model_folder, filename)
                    
                    # 同名ファイルが存在する場合のファイル名調整
                    counter = 1
                    original_destination = destination
                    while os.path.exists(destination):
                        name, ext = os.path.splitext(filename)
                        new_filename = f"{name}_{counter}{ext}"
                        destination = os.path.join(model_folder, new_filename)
                        counter += 1
                    
                    shutil.move(image_path, destination)
                    processed_count += 1
                except Exception as e:
                    continue
        
        messagebox.showinfo("完了", f"処理が完了しました。\n{processed_count}個のファイルを分類しました。")
    
    def start_classification(self):
        """分類処理を別スレッドで開始"""
        self.execute_button.config(state="disabled")
        
        def run_classification():
            try:
                self.classify_images()
            finally:
                self.root.after(0, self.finish_classification)
        
        thread = threading.Thread(target=run_classification)
        thread.daemon = True
        thread.start()
    
    def finish_classification(self):
        """分類処理完了時の処理"""
        self.execute_button.config(state="normal")

def main():
    root = tk.Tk()
    app = JPEGClassifierApp(root)
    root.mainloop()

if __name__ == "__main__":
    main()
