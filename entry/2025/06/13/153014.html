Claude）Python）透過画像作成Windowsアプリ作成。

2025-06-13 · 写真・カメラ ＜AI＞ コンピュータ

Pythonで「透過画像作成」Windowsアプリ作成。画像の白に近い色を透過にする。元データはJPEGまたはPNG。白の範囲は指定できる。AIツールClaudeに作成してもらった。

【使い方】

  • 元画像ファイルは、ドラッグ＆ドロップで指定できる。
  • ドラッグ＆ドロップした画像の下には背景色を表示して、透過状態を確認できる。
  • 背景色は、薄い黄色、薄い青、薄いピンクから選択できる。
  • 透過にする色は白に近い色とする。スライダーで白に近い色の範囲を指定できる。
  • 「保存」ボタンで、元画像と同じフォルダー内に、元画像ファイル名＋"-transparent.png"の名前で保存する。保存ファイル名は変更可能。
【Pythonインストール】

Pythonがインストールされていなければ、インストールする。

Windows）「Python（Ver.3.11.5）」インストール。Windows版Pythonのインストール手順の備忘録。 ■ Pythonダウンロード 現時点の最新、Ver.3.11.5。 ■ Python インストール 「 python-3.11.5-amd64.exe 」を実行。…晴歩雨描 · 2023-09-08

次のライブラリが必要なので、インストールする。

pip install pillow tkinterdnd2 numpy
【Pythonプログラム】

作成されたプログラムを、「透過画像作成.pyw」（名前は任意）という名前にして、デスクトップに置いておき、ダブルクリックで実行する。

【Pythonプログラムコード】
import tkinter as tk
from tkinter import ttk, filedialog, messagebox
from tkinterdnd2 import DND_FILES, TkinterDnD
from PIL import Image, ImageTk, ImageOps
import os
import numpy as np

class ImageTransparencyApp:
    def __init__(self, root):
        self.root = root
        self.root.title("画像透過アプリケーション")
        self.root.geometry("1200x800")
        self.root.configure(bg='#f0f0f0')
        
        # 変数の初期化
        self.original_image = None
        self.processed_image = None
        self.image_path = None
        
        # 背景色の選択肢
        self.background_colors = {
            "薄い黄色": (255, 255, 200),
            "薄い青": (200, 220, 255),
            "薄いピンク": (255, 220, 220)
        }
        
        self.setup_ui()
        self.setup_dnd()
        
    def setup_ui(self):
        # メインフレーム
        main_frame = tk.Frame(self.root, bg='#f0f0f0')
        main_frame.pack(fill=tk.BOTH, expand=True, padx=10, pady=10)
        
        # 左側フレーム（コントロール）
        control_frame = tk.Frame(main_frame, bg='#f0f0f0', width=250)
        control_frame.pack(side=tk.LEFT, fill=tk.Y, padx=(0, 10))
        control_frame.pack_propagate(False)
        
        # ドラッグ&ドロップエリア
        self.drop_label = tk.Label(
            control_frame,
            text="画像ファイルを\nここにドラッグ&ドロップ\n\n(JPEG/PNG対応)",
            bg='#e8e8e8',
            fg='#666666',
            font=('Arial', 12),
            relief=tk.SUNKEN,
            bd=2,
            height=6
        )
        self.drop_label.pack(fill=tk.X, pady=(0, 20))
        
        # 透過設定フレーム
        transparency_frame = tk.LabelFrame(control_frame, text="透過設定", bg='#f0f0f0', font=('Arial', 10, 'bold'))
        transparency_frame.pack(fill=tk.X, pady=(0, 10))
        
        # 透過範囲スライダー（下限）
        tk.Label(transparency_frame, text="透過範囲（下限）:", bg='#f0f0f0').pack(anchor=tk.W, padx=5, pady=(5, 0))
        self.transparency_min = tk.DoubleVar(value=240)
        self.min_scale = tk.Scale(
            transparency_frame,
            from_=100,
            to=255,
            orient=tk.HORIZONTAL,
            variable=self.transparency_min,
            command=self.update_transparency,
            bg='#f0f0f0'
        )
        self.min_scale.pack(fill=tk.X, padx=5, pady=(0, 5))
        
        # 透過範囲スライダー（上限）
        tk.Label(transparency_frame, text="透過範囲（上限）:", bg='#f0f0f0').pack(anchor=tk.W, padx=5)
        self.transparency_max = tk.DoubleVar(value=255)
        self.max_scale = tk.Scale(
            transparency_frame,
            from_=100,
            to=255,
            orient=tk.HORIZONTAL,
            variable=self.transparency_max,
            command=self.update_transparency,
            bg='#f0f0f0'
        )
        self.max_scale.pack(fill=tk.X, padx=5, pady=(0, 10))
        
        # 背景色選択フレーム
        bg_frame = tk.LabelFrame(control_frame, text="背景色", bg='#f0f0f0', font=('Arial', 10, 'bold'))
        bg_frame.pack(fill=tk.X, pady=(0, 10))
        
        self.bg_color_var = tk.StringVar(value="薄い黄色")
        for color_name in self.background_colors.keys():
            tk.Radiobutton(
                bg_frame,
                text=color_name,
                variable=self.bg_color_var,
                value=color_name,
                command=self.update_transparency,
                bg='#f0f0f0'
            ).pack(anchor=tk.W, padx=5, pady=2)
        
        # ボタンフレーム
        button_frame = tk.Frame(control_frame, bg='#f0f0f0')
        button_frame.pack(fill=tk.X, pady=(10, 0))
        
        # ファイル選択ボタン
        self.file_button = tk.Button(
            button_frame,
            text="ファイル選択",
            command=self.select_file,
            bg='#4CAF50',
            fg='white',
            font=('Arial', 10, 'bold'),
            relief=tk.RAISED,
            bd=2
        )
        self.file_button.pack(fill=tk.X, pady=(0, 5))
        
        # 保存ボタン
        self.save_button = tk.Button(
            button_frame,
            text="保存",
            command=self.save_image,
            bg='#2196F3',
            fg='white',
            font=('Arial', 10, 'bold'),
            relief=tk.RAISED,
            bd=2,
            state=tk.DISABLED
        )
        self.save_button.pack(fill=tk.X)
        
        # 右側フレーム（画像表示）
        self.image_frame = tk.Frame(main_frame, bg='white', relief=tk.SUNKEN, bd=2)
        self.image_frame.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True)
        
        # 画像表示ラベル
        self.image_label = tk.Label(
            self.image_frame,
            text="画像がここに表示されます",
            bg='white',
            fg='#999999',
            font=('Arial', 16)
        )
        self.image_label.pack(expand=True)
        
    def setup_dnd(self):
        # ドラッグ&ドロップの設定
        self.drop_label.drop_target_register(DND_FILES)
        self.drop_label.dnd_bind('<<Drop>>', self.on_drop)
        
    def on_drop(self, event):
        # ドロップされたファイルの処理
        files = self.root.tk.splitlist(event.data)
        if files:
            file_path = files[0]
            self.load_image(file_path)
    
    def select_file(self):
        # ファイル選択ダイアログ
        file_path = filedialog.askopenfilename(
            title="画像ファイルを選択",
            filetypes=[
                ("画像ファイル", "*.jpg *.jpeg *.png"),
                ("JPEG", "*.jpg *.jpeg"),
                ("PNG", "*.png"),
                ("すべてのファイル", "*.*")
            ]
        )
        if file_path:
            self.load_image(file_path)
    
    def load_image(self, file_path):
        try:
            # 画像の読み込み
            self.original_image = Image.open(file_path).convert('RGBA')
            self.image_path = file_path
            
            # ドロップエリアのテキストを更新
            filename = os.path.basename(file_path)
            self.drop_label.config(text=f"読み込み完了:\n{filename}")
            
            # 透過処理を実行
            self.update_transparency()
            
            # 保存ボタンを有効化
            self.save_button.config(state=tk.NORMAL)
            
        except Exception as e:
            messagebox.showerror("エラー", f"画像の読み込みに失敗しました:\n{str(e)}")
    
    def update_transparency(self, *args):
        if self.original_image is None:
            return
        
        try:
            # 透過範囲の取得
            min_val = int(self.transparency_min.get())
            max_val = int(self.transparency_max.get())
            
            # 範囲チェック
            if min_val > max_val:
                max_val = min_val
                self.transparency_max.set(max_val)
            
            # 背景色の取得
            bg_color = self.background_colors[self.bg_color_var.get()]
            
            # 透過処理
            img_array = np.array(self.original_image)
            
            # グレースケールに変換して輝度を計算
            if len(img_array.shape) == 3:
                gray = np.dot(img_array[...,:3], [0.299, 0.587, 0.114])
            else:
                gray = img_array[:,:,0]
            
            # 透過マスクの作成（指定した範囲の輝度を透過）
            mask = (gray >= min_val) & (gray <= max_val)
            
            # アルファチャンネルの設定
            alpha = np.where(mask, 0, 255).astype(np.uint8)
            
            # 新しい画像の作成
            result = img_array.copy()
            result[:,:,3] = alpha
            
            # PIL画像に変換
            self.processed_image = Image.fromarray(result, 'RGBA')
            
            # 背景色画像を作成
            bg_image = Image.new('RGB', self.processed_image.size, bg_color)
            
            # 背景色と合成
            display_image = Image.alpha_composite(
                bg_image.convert('RGBA'),
                self.processed_image
            ).convert('RGB')
            
            # 表示用にリサイズ
            self.display_image(display_image)
            
        except Exception as e:
            messagebox.showerror("エラー", f"透過処理に失敗しました:\n{str(e)}")
    
    def display_image(self, image):
        # 画像フレームのサイズを取得
        self.image_frame.update_idletasks()
        frame_width = self.image_frame.winfo_width() - 20
        frame_height = self.image_frame.winfo_height() - 20
        
        if frame_width <= 0 or frame_height <= 0:
            self.root.after(100, lambda: self.display_image(image))
            return
        
        # 画像をフレームに合わせてリサイズ
        img_width, img_height = image.size
        
        # アスペクト比を保持してリサイズ
        ratio = min(frame_width / img_width, frame_height / img_height)
        new_width = int(img_width * ratio)
        new_height = int(img_height * ratio)
        
        if new_width > 0 and new_height > 0:
            display_img = image.resize((new_width, new_height), Image.Resampling.LANCZOS)
            
            # PhotoImageに変換
            photo = ImageTk.PhotoImage(display_img)
            
            # ラベルに画像を設定
            self.image_label.config(image=photo, text="")
            self.image_label.image = photo  # 参照を保持
    
    def save_image(self):
        if self.processed_image is None or self.image_path is None:
            messagebox.showwarning("警告", "保存する画像がありません。")
            return
        
        try:
            # デフォルトの保存ファイル名を生成
            dir_path = os.path.dirname(self.image_path)
            filename = os.path.splitext(os.path.basename(self.image_path))[0]
            default_filename = f"{filename}-transparent.png"
            default_path = os.path.join(dir_path, default_filename)
            
            # 保存ダイアログ
            save_path = filedialog.asksaveasfilename(
                title="透過画像を保存",
                initialdir=dir_path,
                initialfile=default_filename,
                defaultextension=".png",
                filetypes=[
                    ("PNG", "*.png"),
                    ("すべてのファイル", "*.*")
                ]
            )
            
            if save_path:
                # PNG形式で保存
                self.processed_image.save(save_path, "PNG")
                messagebox.showinfo("完了", f"画像を保存しました:\n{save_path}")
                
        except Exception as e:
            messagebox.showerror("エラー", f"画像の保存に失敗しました:\n{str(e)}")

def main():
    # TkinterDnDを使用してルートウィンドウを作成
    root = TkinterDnD.Tk()
    app = ImageTransparencyApp(root)
    root.mainloop()

if __name__ == "__main__":
    main()
