これは、写真の一部をスポット的に編集するアプリ。

これに加えて、クローンコピー修正もできるようにしたバージョンをPythonで作成。

■ スポット修正

写真の不要な部分（ゴミ、シミなど）を自動的に消す機能。

  • 左クリックした場所の周辺の画像を使って修復。
  • AIが自然な感じに仕上げてくれる。
  • 小さな汚れや不要物を消すのに最適。
■ コピー修正（クローンコピー）

写真の一部を別の場所にコピー＆貼り付けする機能。

  • 右クリック：コピーしたい部分を取得。
  • 左クリック：別の場所に貼り付け。
  • 周辺となじむように自動調整される。
  • 同じ模様を増やしたり、特定の部分で隠したい時に便利。
■ 使い方
  • 【画像の読み込み】
    ・画像ファイルをウィンドウにドラッグ&ドロップ
  • 【スポット修正モード】
    ・不要な物を自動的に周りと馴染ませて消す
    ・左クリック：その場所を自動修正
  • 【コピー修正モード】
    ・別の場所の画像をコピーして貼り付ける
    ・右クリック：コピー元を取得
    ・左クリック：取得した画像を貼り付け
  • 【修正サイズ】
    ・スライダーで修正する範囲の大きさを調整
    ・1～100ピクセルの間で設定可能
  • 【表示モード】
    ・ウィンドウに合わせる：画面内に収まるように表示
    ・実寸表示：元の画像サイズで表示
      （実寸表示時はスクロールバーで移動可能）
  • 【操作の取り消し】
    ・一つ戻す：最後の操作を取り消す
    ・すべて戻す：元の画像に戻す
  • 【保存】
    ・編集した画像を保存
    ・元の画像と同じ場所に「ファイル名_編集」で保存

import tkinter as tk
from tkinter import filedialog, messagebox
from tkinterdnd2 import DND_FILES, TkinterDnD
from PIL import Image, ImageTk, ImageDraw, ImageFont
import cv2
import numpy as np
import os

class PhotoSpotEditor:
    def __init__(self, root):
        self.root = root
        self.root.title("写真スポット編集ツール")
        self.root.geometry("1200x800")
        
        # 変数の初期化
        self.original_image = None
        self.display_image = None
        self.current_image = None
        self.tk_image = None
        self.image_path = None
        self.history = []
        self.clone_source = None
        self.clone_display = None
        self.icc_profile = None
        
        # 設定
        self.mode = tk.StringVar(value="spot")
        self.spot_size = tk.IntVar(value=30)
        self.fit_to_window = tk.BooleanVar(value=True)
        self.canvas_x = 0
        self.canvas_y = 0
        self.display_scale = 1.0
        
        # スポットサイズ変更時にコピー元画像表示を更新
        self.spot_size.trace('w', lambda *args: self.update_clone_display())
        
        self.setup_ui()
        
    def setup_ui(self):
        main_frame = tk.Frame(self.root)
        main_frame.pack(fill=tk.BOTH, expand=True)
        
        # 左側コントロールパネル
        control_frame = tk.Frame(main_frame, width=250, bg="#f0f0f0")
        control_frame.pack(side=tk.LEFT, fill=tk.Y, padx=5, pady=5)
        control_frame.pack_propagate(False)
        
        # タイトル
        title_label = tk.Label(control_frame, text="写真スポット編集", 
                              font=("Arial", 14, "bold"), bg="#f0f0f0")
        title_label.pack(pady=10)
        
        # モード選択
        mode_frame = tk.LabelFrame(control_frame, text="修正モード", bg="#f0f0f0")
        mode_frame.pack(fill=tk.X, padx=10, pady=5)
        
        tk.Radiobutton(mode_frame, text="スポット修正", variable=self.mode, 
                      value="spot", bg="#f0f0f0").pack(anchor=tk.W, padx=5)
        tk.Radiobutton(mode_frame, text="コピー修正", variable=self.mode, 
                      value="clone", bg="#f0f0f0").pack(anchor=tk.W, padx=5)
        
        # スポットサイズ
        size_frame = tk.LabelFrame(control_frame, text="修正サイズ", bg="#f0f0f0")
        size_frame.pack(fill=tk.X, padx=10, pady=5)
        
        size_label = tk.Label(size_frame, text="サイズ: 30 px", bg="#f0f0f0")
        size_label.pack()
        
        size_slider = tk.Scale(size_frame, from_=1, to=100, orient=tk.HORIZONTAL,
                              variable=self.spot_size, bg="#f0f0f0",
                              command=lambda v: size_label.config(text=f"サイズ: {v} px"))
        size_slider.pack(fill=tk.X, padx=5)
        
        # コピー元画像表示（円形専用）
        clone_frame = tk.LabelFrame(control_frame, text="コピー元画像", bg="#f0f0f0")
        clone_frame.pack(fill=tk.X, padx=10, pady=5)
        
        # Canvasで円形表示を実装
        self.clone_canvas = tk.Canvas(clone_frame, width=140, height=140, 
                                     bg="white", relief=tk.SUNKEN, bd=2)
        self.clone_canvas.pack(padx=5, pady=5)
        self.clone_canvas.create_text(70, 70, text="右クリックで取得", 
                                     fill="gray", tags="placeholder")
        
        # 表示モード
        display_frame = tk.LabelFrame(control_frame, text="表示モード", bg="#f0f0f0")
        display_frame.pack(fill=tk.X, padx=10, pady=5)
        
        tk.Radiobutton(display_frame, text="ウィンドウに合わせる", 
                      variable=self.fit_to_window, value=True, 
                      bg="#f0f0f0", command=self.update_display).pack(anchor=tk.W, padx=5)
        tk.Radiobutton(display_frame, text="実寸表示", 
                      variable=self.fit_to_window, value=False, 
                      bg="#f0f0f0", command=self.update_display).pack(anchor=tk.W, padx=5)
        
        # 操作ボタン
        button_frame = tk.Frame(control_frame, bg="#f0f0f0")
        button_frame.pack(fill=tk.X, padx=10, pady=10)
        
        tk.Button(button_frame, text="一つ戻す", command=self.undo,
                 bg="#4CAF50", fg="white", width=20).pack(pady=2)
        tk.Button(button_frame, text="すべて戻す", command=self.reset_all,
                 bg="#FF9800", fg="white", width=20).pack(pady=2)
        tk.Button(button_frame, text="保存", command=self.save_image,
                 bg="#2196F3", fg="white", width=20).pack(pady=2)
        
        # 使用方法ボタン
        help_btn = tk.Button(control_frame, text="使い方", command=self.show_help,
                            bg="#9E9E9E", fg="white")
        help_btn.pack(side=tk.BOTTOM, pady=10)
        
        # 右側キャンバスエリア
        canvas_frame = tk.Frame(main_frame)
        canvas_frame.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=5, pady=5)
        
        # スクロールバー
        self.h_scrollbar = tk.Scrollbar(canvas_frame, orient=tk.HORIZONTAL)
        self.v_scrollbar = tk.Scrollbar(canvas_frame, orient=tk.VERTICAL)
        
        # キャンバス
        self.canvas = tk.Canvas(canvas_frame, bg="#e0e0e0", cursor="none",
                               xscrollcommand=self.h_scrollbar.set,
                               yscrollcommand=self.v_scrollbar.set)
        
        self.h_scrollbar.config(command=self.canvas.xview)
        self.v_scrollbar.config(command=self.canvas.yview)
        
        # グリッド配置
        self.canvas.grid(row=0, column=0, sticky='nsew')
        canvas_frame.grid_rowconfigure(0, weight=1)
        canvas_frame.grid_columnconfigure(0, weight=1)
        
        # ドラッグ&ドロップ設定
        self.canvas.drop_target_register(DND_FILES)
        self.canvas.dnd_bind('<<Drop>>', self.on_drop)
        
        # マウスイベント
        self.canvas.bind("<Motion>", self.on_mouse_move)
        self.canvas.bind("<Button-1>", self.on_left_click)
        self.canvas.bind("<Button-3>", self.on_right_click)
        self.canvas.bind("<Leave>", self.on_mouse_leave)
        self.canvas.bind("<Configure>", self.on_canvas_resize)
        
        # 初期メッセージ
        self.canvas.create_text(400, 300, text="画像をここにドラッグ&ドロップ", 
                               font=("Arial", 16), fill="gray", tags="placeholder")
    
    def on_canvas_resize(self, event):
        if self.current_image is not None and self.fit_to_window.get():
            self.update_display()
    
    def on_drop(self, event):
        file_path = event.data.strip('{}')
        if file_path.lower().endswith(('.png', '.jpg', '.jpeg', '.bmp', '.tiff')):
            self.load_image(file_path)
    
    def load_image(self, path):
        try:
            self.image_path = path
            pil_img = Image.open(path)
            
            # ICCプロファイルを保持（色合い保持のため）
            self.icc_profile = pil_img.info.get('icc_profile', None)
            
            if pil_img.mode != 'RGB':
                pil_img = pil_img.convert('RGB')
            self.original_image = np.array(pil_img)
            self.current_image = self.original_image.copy()
            self.history = [self.current_image.copy()]
            self.clone_source = None
            self.update_clone_display()
            self.update_display()
            self.canvas.delete("placeholder")
        except Exception as e:
            messagebox.showerror("エラー", f"画像の読み込みに失敗しました:\n{str(e)}")
    
    def update_display(self):
        if self.current_image is None:
            return
        
        canvas_width = self.canvas.winfo_width()
        canvas_height = self.canvas.winfo_height()
        
        if canvas_width <= 1 or canvas_height <= 1:
            self.root.after(100, self.update_display)
            return
        
        img_height, img_width = self.current_image.shape[:2]
        
        if self.fit_to_window.get():
            # ウィンドウに合わせる
            scale_w = canvas_width / img_width
            scale_h = canvas_height / img_height
            self.display_scale = min(scale_w, scale_h, 1.0)
            
            # スクロールバーを非表示
            self.h_scrollbar.grid_forget()
            self.v_scrollbar.grid_forget()
            
            # 中央配置
            new_width = int(img_width * self.display_scale)
            new_height = int(img_height * self.display_scale)
            self.canvas_x = (canvas_width - new_width) // 2
            self.canvas_y = (canvas_height - new_height) // 2
        else:
            # 実寸表示
            self.display_scale = 1.0
            
            # スクロールバーを表示
            self.h_scrollbar.grid(row=1, column=0, sticky='ew')
            self.v_scrollbar.grid(row=0, column=1, sticky='ns')
            
            # 左上配置
            self.canvas_x = 0
            self.canvas_y = 0
        
        new_width = int(img_width * self.display_scale)
        new_height = int(img_height * self.display_scale)
        
        self.display_image = cv2.resize(self.current_image, (new_width, new_height), 
                                       interpolation=cv2.INTER_AREA)
        
        # PIL画像に変換
        pil_image = Image.fromarray(self.display_image)
        self.tk_image = ImageTk.PhotoImage(pil_image)
        
        self.canvas.delete("all")
        self.canvas.create_image(self.canvas_x, self.canvas_y, 
                                anchor=tk.NW, image=self.tk_image, tags="image")
        
        # スクロール領域の設定
        if not self.fit_to_window.get():
            self.canvas.config(scrollregion=(0, 0, new_width, new_height))
        else:
            self.canvas.config(scrollregion=(0, 0, canvas_width, canvas_height))
    
    def get_image_coords(self, event_x, event_y):
        """イベント座標を画像座標に変換（スクロール対応）"""
        if self.fit_to_window.get():
            # ウィンドウに合わせるモード
            x = int((event_x - self.canvas_x) / self.display_scale)
            y = int((event_y - self.canvas_y) / self.display_scale)
        else:
            # 実寸表示モード（スクロール考慮）
            canvas_x = self.canvas.canvasx(event_x)
            canvas_y = self.canvas.canvasy(event_y)
            x = int((canvas_x - self.canvas_x) / self.display_scale)
            y = int((canvas_y - self.canvas_y) / self.display_scale)
        return x, y
    
    def on_mouse_move(self, event):
        if self.display_image is None:
            return
        
        # スクロールを考慮した座標取得
        if self.fit_to_window.get():
            canvas_x = event.x
            canvas_y = event.y
        else:
            canvas_x = self.canvas.canvasx(event.x)
            canvas_y = self.canvas.canvasy(event.y)
        
        x = int((canvas_x - self.canvas_x) / self.display_scale)
        y = int((canvas_y - self.canvas_y) / self.display_scale)
        
        if 0 <= x < self.current_image.shape[1] and 0 <= y < self.current_image.shape[0]:
            self.canvas.delete("cursor")
            radius = int(self.spot_size.get() * self.display_scale / 2)
            # スクロール位置を考慮した表示座標
            display_x = self.canvas_x + x * self.display_scale
            display_y = self.canvas_y + y * self.display_scale
            self.canvas.create_oval(display_x - radius, display_y - radius,
                                   display_x + radius, display_y + radius,
                                   outline="red", width=2, tags="cursor")
    
    def on_mouse_leave(self, event):
        self.canvas.delete("cursor")
    
    def on_left_click(self, event):
        if self.current_image is None:
            return
        
        x, y = self.get_image_coords(event.x, event.y)
        
        if self.mode.get() == "spot":
            self.apply_spot_healing(x, y)
        else:
            self.apply_clone_stamp(x, y)
    
    def on_right_click(self, event):
        if self.current_image is None or self.mode.get() != "clone":
            return
        
        x, y = self.get_image_coords(event.x, event.y)
        
        radius = self.spot_size.get() // 2
        h, w = self.current_image.shape[:2]
        
        if x - radius < 0 or y - radius < 0 or x + radius >= w or y + radius >= h:
            messagebox.showwarning("警告", "画像の端に近すぎます")
            return
        
        y1, y2 = y - radius, y + radius
        x1, x2 = x - radius, x + radius
        self.clone_source = self.current_image[y1:y2, x1:x2].copy()
        
        self.update_clone_display()
    
    def update_clone_display(self):
        """Canvasに円形でクローン元を描画"""
        self.clone_canvas.delete("all")
        
        if self.clone_source is not None:
            size = self.spot_size.get()
            
            # 表示サイズの計算
            max_display = 120
            if size > max_display:
                display_size = max_display
            else:
                display_size = size
            
            # クローン元画像をPILに変換
            clone_pil = Image.fromarray(self.clone_source)
            if clone_pil.size != (display_size, display_size):
                clone_pil = clone_pil.resize((display_size, display_size), Image.LANCZOS)
            
            # 円形マスクを作成
            mask = Image.new('L', (display_size, display_size), 0)
            draw = ImageDraw.Draw(mask)
            draw.ellipse([0, 0, display_size-1, display_size-1], fill=255)
            
            # 背景画像（灰色）
            background = Image.new('RGB', (display_size, display_size), (240, 240, 240))
            
            # マスクを使って円形に切り抜き
            clone_pil_rgb = clone_pil.convert('RGB')
            result = Image.composite(clone_pil_rgb, background, mask)
            
            # Tkinter用に変換
            self.clone_display = ImageTk.PhotoImage(result)
            
            # Canvasの中央に配置
            center_x = 70
            center_y = 70
            self.clone_canvas.create_image(center_x, center_y, image=self.clone_display)
            
            # サイズ表示
            self.clone_canvas.create_text(70, 10, text=f"{size}px", 
                                         fill="black", font=("Arial", 10, "bold"))
        else:
            self.clone_canvas.create_text(70, 70, text="右クリックで取得", 
                                         fill="gray", tags="placeholder")
    
    def apply_spot_healing(self, x, y):
        radius = self.spot_size.get() // 2
        h, w = self.current_image.shape[:2]
        
        if x - radius < 0 or y - radius < 0 or x + radius >= w or y + radius >= h:
            return
        
        self.history.append(self.current_image.copy())
        
        mask = np.zeros((h, w), dtype=np.uint8)
        cv2.circle(mask, (x, y), radius, 255, -1)
        
        self.current_image = cv2.inpaint(self.current_image, mask, 
                                        radius, cv2.INPAINT_TELEA)
        
        self.update_display()
    
    def apply_clone_stamp(self, x, y):
        if self.clone_source is None:
            messagebox.showinfo("情報", "まず右クリックでコピー元を取得してください")
            return
        
        radius = self.spot_size.get() // 2
        h, w = self.current_image.shape[:2]
        
        if x - radius < 0 or y - radius < 0 or x + radius >= w or y + radius >= h:
            return
        
        self.history.append(self.current_image.copy())
        
        y1, y2 = y - radius, y + radius
        x1, x2 = x - radius, x + radius
        
        clone_h, clone_w = self.clone_source.shape[:2]
        target_h, target_w = y2 - y1, x2 - x1
        
        if clone_h != target_h or clone_w != target_w:
            clone_to_paste = cv2.resize(self.clone_source, (target_w, target_h), 
                                       interpolation=cv2.INTER_CUBIC)
        else:
            clone_to_paste = self.clone_source.copy()
        
        mask = np.zeros((target_h, target_w), dtype=np.uint8)
        cv2.circle(mask, (radius, radius), radius, 255, -1)
        
        blur_size = max(3, radius // 3)
        if blur_size % 2 == 0:
            blur_size += 1
        mask_blurred = cv2.GaussianBlur(mask, (blur_size, blur_size), 0)
        
        mask_normalized = mask_blurred.astype(float) / 255.0
        mask_3ch = np.stack([mask_normalized] * 3, axis=2)
        
        result = (clone_to_paste * mask_3ch + 
                 self.current_image[y1:y2, x1:x2] * (1 - mask_3ch)).astype(np.uint8)
        
        self.current_image[y1:y2, x1:x2] = result
        
        self.update_display()
    
    def undo(self):
        if len(self.history) > 1:
            self.history.pop()
            self.current_image = self.history[-1].copy()
            self.update_display()
        else:
            messagebox.showinfo("情報", "これ以上戻せません")
    
    def reset_all(self):
        if self.original_image is not None:
            self.current_image = self.original_image.copy()
            self.history = [self.current_image.copy()]
            self.update_display()
    
    def save_image(self):
        if self.current_image is None:
            messagebox.showwarning("警告", "保存する画像がありません")
            return
        
        # 拡張子の決定と初期ファイル名（拡張子なし）の設定
        if self.image_path:
            directory = os.path.dirname(self.image_path)
            basename = os.path.basename(self.image_path)
            name, ext = os.path.splitext(basename)
            
            ext_lower = ext.lower()
            
            # デフォルト拡張子と filetypes の順序を決定
            if ext_lower in ('.jpg', '.jpeg'):
                default_ext = '.jpg'
                filetypes_list = [("JPEG画像", "*.jpg"), ("PNG画像", "*.png"), ("すべてのファイル", "*.*")]
            else:
                default_ext = '.png'
                filetypes_list = [("PNG画像", "*.png"), ("JPEG画像", "*.jpg"), ("すべてのファイル", "*.*")]

            try:
                # 拡張子を除いたファイル名を設定
                initial_filename_no_ext = f"{name}_編集"
            except:
                initial_filename_no_ext = f"{name}_edited"
        else:
            directory = os.path.expanduser("~")
            default_ext = ".png"
            initial_filename_no_ext = "編集済み画像" # 拡張子なし
            filetypes_list = [("PNG画像", "*.png"), ("JPEG画像", "*.jpg"), ("すべてのファイル", "*.*")]

        
        # initialfile には拡張子を付けずに渡す
        filepath = filedialog.asksaveasfilename(
            initialdir=directory,
            initialfile=initial_filename_no_ext, # ここを修正
            defaultextension=default_ext,
            filetypes=filetypes_list
        )
        
        if filepath:
            try:
                img_pil = Image.fromarray(self.current_image)
                filepath_lower = filepath.lower()
                
                # ICCプロファイルの設定（色合い保持のため）
                save_kwargs = {}
                if self.icc_profile:
                    save_kwargs['icc_profile'] = self.icc_profile
                
                # ユーザーが選択した拡張子に基づいて保存形式を決定
                if filepath_lower.endswith(('.jpg', '.jpeg')):
                    img_pil.save(filepath, 'JPEG', quality=95, **save_kwargs)
                elif filepath_lower.endswith('.png'):
                    img_pil.save(filepath, 'PNG', **save_kwargs)
                else:
                    # ユーザーが拡張子を指定しなかった場合、defaultextensionで補完する
