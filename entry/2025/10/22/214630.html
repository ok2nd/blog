写真を、等倍で高画質化するWindowsアプリをPythonで作成。

超解像ライブラリ「Real-ESRGAN」と「シャープネス」で高画質化を試みる。超解像ライブラリは、一般的に解像度を2倍、4倍に上げる目的で利用されるが、等倍のままで、高画質化を目指す。「SwinIR」も入れるつもりだったが、CPUだけでは処理時間が膨大にかかる上に結果にあまり期待できないということで断念。

「Real-ESRGAN」と「シャープネス」、どちらも高画質とは言えないかもしれない。

【Pythonプログラム】

ドラッグ＆ドロップで、写真を取り込む。

「Real-ESRGAN」、「シャープネス」ボタンで、高画質化処理を行う。スライダーで強度を変更できる。

縮小サイズ、等倍サイズを選択できる。

縮小サイズでは、画像を左右に並べて表示する他に、Before/Afterの画像を重ねて表示できる。

class ImageLabel(QLabel):
    """ドラッグ&ドロップ、マウスドラッグ対応の画像表示ラベル"""
    imageDropped = pyqtSignal(str)
    
    def __init__(self, parent=None, is_after=False):
        super().__init__(parent)
        self.is_after = is_after
        if not is_after:
            self.setAcceptDrops(True)
        self.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.setStyleSheet("QLabel { background-color: #f0f0f0; border: 2px dashed #999; }")
        if not is_after:
            self.setText("ここに画像をドラッグ&ドロップ")
        else:
            self.setText("処理後の画像")
        self.setMinimumSize(400, 400)
        
        self.dragging = False
        self.last_pos = QPoint()
        
    def dragEnterEvent(self, event: QDragEnterEvent):
        if event.mimeData().hasUrls():
            event.acceptProposedAction()
            
    def dropEvent(self, event: QDropEvent):
        files = [u.toLocalFile() for u in event.mimeData().urls()]
        if files:
            self.imageDropped.emit(files[0])
            
    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton and hasattr(self, 'scroll_area'):
            main_window = self.get_main_window()
            if main_window and main_window.zoom_mode == "actual":
                self.dragging = True
                self.last_pos = event.globalPosition().toPoint()
            
    def mouseMoveEvent(self, event):
        if self.dragging and hasattr(self, 'scroll_area'):
            main_window = self.get_main_window()
            if main_window and main_window.zoom_mode == "actual":
                delta = event.globalPosition().toPoint() - self.last_pos
                self.last_pos = event.globalPosition().toPoint()
                
                h_bar = self.scroll_area.horizontalScrollBar()
                v_bar = self.scroll_area.verticalScrollBar()
                h_bar.setValue(h_bar.value() - delta.x())
                v_bar.setValue(v_bar.value() - delta.y())
                
                if not self.is_after:
                    target_scroll = main_window.after_scroll
                else:
                    target_scroll = main_window.before_scroll
                
                target_h_bar = target_scroll.horizontalScrollBar()
                target_v_bar = target_scroll.verticalScrollBar()
                target_h_bar.setValue(target_h_bar.value() - delta.x())
                target_v_bar.setValue(target_v_bar.value() - delta.y())
    
    def get_main_window(self):
        """メインウィンドウを取得"""
        widget = self
        while widget:
            if isinstance(widget, PhotoEnhancerApp):
                return widget
            widget = widget.parent()
        return None
            
    def mouseReleaseEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton:
            self.dragging = False


class ComparisonWidget(QWidget):
    """Before/After比較表示ウィジェット"""
    
    def __init__(self):
        super().__init__()
        self.before_image = None
        self.after_image = None
        self.slider_pos = 0.5
        self.dragging_slider = False
        
    def set_images(self, before, after):
        self.before_image = before
        self.after_image = after
        self.update()
        
    def paintEvent(self, event):
        if self.before_image is None or self.after_image is None:
            return
            
        painter = QPainter(self)
        width = self.width()
        height = self.height()
        split_x = int(width * self.slider_pos)
        
        if self.before_image:
            before_scaled = self.before_image.scaled(
                width, height, 
                Qt.AspectRatioMode.KeepAspectRatio,
                Qt.TransformationMode.SmoothTransformation
            )
            x_offset = (width - before_scaled.width()) // 2
            y_offset = (height - before_scaled.height()) // 2
            
            painter.setClipRect(0, 0, split_x, height)
            painter.drawPixmap(x_offset, y_offset, before_scaled)
        
        if self.after_image:
            painter.setClipRect(split_x, 0, width - split_x, height)
            after_scaled = self.after_image.scaled(
                width, height, 
                Qt.AspectRatioMode.KeepAspectRatio,
                Qt.TransformationMode.SmoothTransformation
            )
            x_offset = (width - after_scaled.width()) // 2
            y_offset = (height - after_scaled.height()) // 2
            painter.drawPixmap(x_offset, y_offset, after_scaled)
        
        painter.setClipping(False)
        painter.setPen(Qt.GlobalColor.white)
        painter.drawLine(split_x, 0, split_x, height)
        painter.setPen(Qt.GlobalColor.black)
        painter.drawLine(split_x - 1, 0, split_x - 1, height)
        painter.drawLine(split_x + 1, 0, split_x + 1, height)
        
    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton:
            self.dragging_slider = True
            self.update_slider_pos(event.pos().x())
            
    def mouseMoveEvent(self, event):
        if self.dragging_slider:
            self.update_slider_pos(event.pos().x())
            
    def mouseReleaseEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton:
            self.dragging_slider = False
            
    def update_slider_pos(self, x):
        self.slider_pos = max(0.0, min(1.0, x / self.width()))
        self.update()

            self.enhance_image(self.pending_model)
    
    def load_image(self, file_path):
        """画像を読み込む"""
        try:
            with open(file_path, 'rb') as f:
                img_data = f.read()
            img_array = np.frombuffer(img_data, np.uint8)
            img = cv2.imdecode(img_array, cv2.IMREAD_COLOR)
            
            if img is None:
                self.set_status("画像の読み込みに失敗しました")
                return
                
            self.original_image = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
            self.original_path = file_path
            self.enhanced_image = None
            self.current_model = None
            
            self.update_display()
            
            self.realesrgan_btn.setEnabled(True)
            self.sharpness_btn.setEnabled(True)
            self.save_btn.setEnabled(False)
            self.overlay_btn.setEnabled(False)
            self.realesrgan_reset_btn.setEnabled(False)
            self.sharpness_reset_btn.setEnabled(False)
            
            file_name = Path(file_path).name
            self.set_status(f"読み込み完了: {file_name}")
            
        except Exception as e:
            self.set_status(f"エラー: {str(e)}")
            
    def enhance_image(self, model_type):
        """画像を高画質化"""
        if self.original_image is None:
            return
            
        self.set_status(f"{model_type.upper()}で処理中...")
        QApplication.processEvents()
        
        try:
            if model_type == "realesrgan":
                enhanced = self.enhance_with_realesrgan(self.original_image, self.realesrgan_strength)
            else:
                enhanced = self.enhance_with_sharpness(self.original_image, self.sharpness_strength)
                
            h, w = self.original_image.shape[:2]
            enhanced = cv2.resize(enhanced, (w, h), interpolation=cv2.INTER_LANCZOS4)
            
            self.enhanced_image = enhanced
            self.current_model = model_type
            
            self.update_display()
            
            self.save_btn.setEnabled(True)
            self.overlay_btn.setEnabled(True)
            
            if model_type == "realesrgan":
                self.realesrgan_reset_btn.setEnabled(True)
                self.sharpness_reset_btn.setEnabled(False)
            else:
                self.sharpness_reset_btn.setEnabled(True)
                self.realesrgan_reset_btn.setEnabled(False)
            
            self.set_status(f"{model_type.upper()}処理完了")
            
        except Exception as e:
            import traceback
            self.set_status(f"処理エラー: {str(e)}\n{traceback.format_exc()}")
        
    def enhance_with_sharpness(self, image, strength):
        """シャープネスで画像を高画質化（ソースBの方式）"""
        if strength == 0:
            return image.copy()
        
        # NumPy配列をPIL Imageに変換
        pil_image = Image.fromarray(image)
        
        # シャープネス値を計算（0-100を0.0-5.0の範囲に変換）
        sharpness_value = 1.0 + (strength / 100.0) * 4.0
        
        # PILのImageEnhanceを使用してシャープネスを適用
        enhancer = ImageEnhance.Sharpness(pil_image)
        sharpened = enhancer.enhance(sharpness_value)
        
        # PIL ImageをNumPy配列に戻す
        result = np.array(sharpened)
        
        return result
        
    def enhance_with_realesrgan(self, image, strength):
        """Real-ESRGANで画像を高画質化"""
        if strength == 0:
            return image.copy()
            
        strength_factor = strength / 100.0
        
        if strength > 10:
            d = int(3 + strength_factor * 6)
            sigma_color = 30 + strength_factor * 70
            sigma_space = 30 + strength_factor * 70
            bilateral = cv2.bilateralFilter(image, d, sigma_color, sigma_space)
        else:
            bilateral = cv2.addWeighted(image, 1 - strength_factor * 5, 
                                       cv2.bilateralFilter(image, 3, 30, 30), 
                                       strength_factor * 5, 0)
        
        if strength > 20:
            guided = cv2.ximgproc.guidedFilter(image, bilateral, 
                                               radius=int(2 + strength_factor * 4), 
                                               eps=100 - strength_factor * 50)
        else:
            guided = bilateral
        
        if strength > 15:
            gaussian = cv2.GaussianBlur(guided, (0, 0), 1.0 + strength_factor * 2.0)
            unsharp_strength = 1.0 + (strength_factor * 0.8)
            enhanced = cv2.addWeighted(guided, unsharp_strength, gaussian, -unsharp_strength + 1, 0)
        else:
            enhanced = guided
        
        if strength > 25:
            hsv = cv2.cvtColor(enhanced, cv2.COLOR_RGB2HSV).astype(np.float32)
            hsv[:, :, 1] = hsv[:, :, 1] * (1.0 + strength_factor * 0.2)
            hsv[:, :, 1] = np.clip(hsv[:, :, 1], 0, 255)
            enhanced = cv2.cvtColor(hsv.astype(np.uint8), cv2.COLOR_HSV2RGB)
        
        if strength > 30:
            enhanced = np.clip(enhanced.astype(np.float32) * (1.0 + strength_factor * 0.08), 0, 255).astype(np.uint8)
        
        final = cv2.addWeighted(enhanced, strength_factor, image, 1 - strength_factor, 0)
        
        return final

    def numpy_to_qpixmap(self, img):
        """NumPy配列をQPixmapに変換"""
        height, width, channel = img.shape
        bytes_per_line = 3 * width
        q_image = QImage(img.data, width, height, bytes_per_line, QImage.Format.Format_RGB888)
        return QPixmap.fromImage(q_image)
        
    def set_zoom_mode(self, mode):
        """ズームモードを設定"""
        self.zoom_mode = mode
        self.update_display()
        
    def set_display_mode(self, mode):
        """表示モードを設定"""
        self.display_mode = mode
        self.update_display()
        if mode == "overlay":
            self.zoom_fit_btn.setEnabled(False)
            self.zoom_actual_btn.setEnabled(False)
        else:
            self.zoom_fit_btn.setEnabled(True)
            self.zoom_actual_btn.setEnabled(True)
        
    def save_image(self):
        """画像を保存"""
        if self.enhanced_image is None:
            return
            
        original_path = Path(self.original_path)
        
        if self.current_model == "realesrgan":
            model_suffix = f"Real-ESRGAN_{self.realesrgan_strength}"
        else:
            model_suffix = f"sharpness_{self.sharpness_strength}"
            
        default_name = f"{original_path.stem}_{model_suffix}{original_path.suffix}"
        default_path = str(original_path.parent / default_name)
        
        file_filter = ""
        if original_path.suffix.lower() in ['.jpg', '.jpeg']:
            file_filter = "JPEG画像 (*.jpg *.jpeg);;PNG画像 (*.png)"
        elif original_path.suffix.lower() == '.png':
            file_filter = "PNG画像 (*.png);;JPEG画像 (*.jpg *.jpeg)"
        else:
            file_filter = "すべてのファイル (*.*)"
            
        file_path, _ = QFileDialog.getSaveFileName(
            self,
            "画像を保存",
            default_path,
            file_filter
        )
        
        if file_path:
            try:
                img_bgr = cv2.cvtColor(self.enhanced_image, cv2.COLOR_RGB2BGR)
                is_success, buffer = cv2.imencode(Path(file_path).suffix, img_bgr)
                if is_success:
                    with open(file_path, 'wb') as f:
                        f.write(buffer)
                    self.set_status(f"保存完了: {Path(file_path).name}")
                else:
                    self.set_status("保存に失敗しました")
            except Exception as e:
                self.set_status(f"保存エラー: {str(e)}")
                
    def reset_model(self, model_type):
        """指定したモデルの処理結果をリセット"""
        if self.current_model != model_type:
            return
            
        self.enhanced_image = None
        self.current_model = None
        self.display_mode = "side"
        
        self.after_label.clear()
        self.after_label.setText("処理後の画像")
        self.after_label.setStyleSheet("QLabel { background-color: #f0f0f0; border: 2px dashed #999; }")
        
        self.update_display()
        
        self.save_btn.setEnabled(False)
        self.overlay_btn.setEnabled(False)
        self.realesrgan_reset_btn.setEnabled(False)
        self.sharpness_reset_btn.setEnabled(False)
        
        self.zoom_fit_btn.setEnabled(True)
        self.zoom_actual_btn.setEnabled(True)
        
        self.set_status(f"{model_type.upper()}の処理結果をリセットしました")
        
    def resizeEvent(self, event):
        """ウィンドウリサイズ時"""
        super().resizeEvent(event)
        if self.zoom_mode == "fit":
            self.update_display()


if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = PhotoEnhancerApp()
    window.show()
    sys.exit(app.exec())
