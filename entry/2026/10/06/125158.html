■１． SQLite データベース：テーブル作成

データベース：data/住所録管理.db

-- 住所録
CREATE TABLE IF NOT EXISTS t_address (
    id_abook INTEGER PRIMARY KEY,
    id_category INTEGER,
    c_name1 TEXT,
    c_name2 TEXT,
    c_yomi1 TEXT,
    c_yomi2 TEXT,
    c_renmei TEXT,
    c_zip1 TEXT,
    c_zip2 TEXT,
    c_kenid TEXT,
    c_address1 TEXT,
    c_address2 TEXT,
    c_phone1 TEXT,
    c_phone2 TEXT,
    c_email TEXT,
    c_memo TEXT,
    c_check1 INTEGER,
    c_check2 INTEGER,
    c_check3 INTEGER,
    c_check4 INTEGER,
    c_check5 INTEGER,
    c_check6 INTEGER,
    c_check7 INTEGER,
    c_check8 INTEGER
);

-- 分類
CREATE TABLE IF NOT EXISTS t_category (
    id_category INTEGER PRIMARY KEY,
    c_categoryName TEXT,
    c_categoryDisplayOrder INTEGER
);

-- 都道府県
CREATE TABLE IF NOT EXISTS t_kenmei (
    c_kenid TEXT PRIMARY KEY,
    c_kenmei TEXT,
    c_area TEXT
);

-- チェックカラムキャプション
CREATE TABLE t_check_caption (
    c_caption1 TEXT,
    c_caption2 TEXT,
    c_caption3 TEXT,
    c_caption4 TEXT,
    c_caption5 TEXT,
    c_caption6 TEXT,
    c_caption7 TEXT,
    c_caption8 TEXT
);

-- 住所録ビュー
CREATE VIEW IF NOT EXISTS v_address AS
SELECT
    a.id_abook,
    a.id_category,
    a.c_name1,
    a.c_name2,
    a.c_yomi1,
    a.c_yomi2,
    a.c_renmei,
    a.c_zip1,
    a.c_zip2,
    a.c_kenid,
    a.c_address1,
    a.c_address2,
    a.c_phone1,
    a.c_phone2,
    a.c_email,
    a.c_memo,
    a.c_check1,
    a.c_check2,
    a.c_check3,
    a.c_check4,
    a.c_check5,
    a.c_check6,
    a.c_check7,
    a.c_check8,
    k.c_kenmei,
    c.c_categoryName,
    c.c_categoryDisplayOrder
FROM t_address AS a
LEFT JOIN t_kenmei AS k
    ON a.c_kenid = k.c_kenid
LEFT JOIN t_category AS c
    ON a.id_category = c.id_category;
