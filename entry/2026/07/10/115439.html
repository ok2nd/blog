# -*- coding: utf-8 -*-
"""
tts_injector_gui.py

指定フォルダ以下の .html / .htm を再帰的に検索し、
読み上げ機能（Web Speech API + 自動スクロール）を注入したコピーを
「reading」フォルダ以下に、元と同じフォルダ構成・同じファイル名で出力するGUIアプリ。

必要ライブラリ:
    pip install beautifulsoup4

実行:
    python tts_injector_gui.py
"""

import os
import sys
import threading
import traceback
from pathlib import Path

import tkinter as tk
from tkinter import filedialog, messagebox, scrolledtext, ttk

from tts_core import inject_tts_file

READING_DIRNAME = "reading"


class TtsInjectorApp:
    def __init__(self, root: tk.Tk):
        self.root = root
        root.title("読み上げ機能 一括注入ツール")
        root.geometry("720x520")
        root.minsize(600, 420)

        self.folder_var = tk.StringVar()
        self.recursive_var = tk.BooleanVar(value=True)
        self.overwrite_var = tk.BooleanVar(value=True)
        self.is_running = False

        self._build_widgets()

    # ---------- UI構築 ----------
    def _build_widgets(self):
        pad = {"padx": 8, "pady": 6}

        frame_top = ttk.Frame(self.root)
        frame_top.pack(fill="x", **pad)

        ttk.Label(frame_top, text="対象フォルダ:").pack(side="left")
        entry = ttk.Entry(frame_top, textvariable=self.folder_var)
        entry.pack(side="left", fill="x", expand=True, padx=6)
        ttk.Button(frame_top, text="参照...", command=self.choose_folder).pack(side="left")

        frame_opts = ttk.Frame(self.root)
        frame_opts.pack(fill="x", **pad)
        ttk.Checkbutton(
            frame_opts, text="サブフォルダも再帰的に処理する", variable=self.recursive_var
        ).pack(side="left")
        ttk.Checkbutton(
            frame_opts, text="出力先が既に存在する場合は上書きする", variable=self.overwrite_var
        ).pack(side="left", padx=12)

        frame_btn = ttk.Frame(self.root)
        frame_btn.pack(fill="x", **pad)
        self.run_button = ttk.Button(
            frame_btn, text="読み上げ機能を注入して 'reading' フォルダへ出力", command=self.start_process
        )
        self.run_button.pack(side="left")

        self.progress = ttk.Progressbar(frame_btn, mode="determinate")
        self.progress.pack(side="left", fill="x", expand=True, padx=10)

        ttk.Label(self.root, text="ログ:").pack(anchor="w", padx=8)
        self.log_box = scrolledtext.ScrolledText(self.root, height=20, state="disabled", wrap="word")
        self.log_box.pack(fill="both", expand=True, padx=8, pady=(0, 8))

    # ---------- ユーティリティ ----------
    def log(self, msg: str):
        def _append():
            self.log_box.configure(state="normal")
            self.log_box.insert("end", msg + "\n")
            self.log_box.see("end")
            self.log_box.configure(state="disabled")
        self.root.after(0, _append)

    def choose_folder(self):
        path = filedialog.askdirectory(title="HTMLファイルがあるフォルダを選択")
        if path:
            self.folder_var.set(path)

    # ---------- 処理本体 ----------
    def find_html_files(self, root_dir: Path, recursive: bool):
        pattern_ext = (".html", ".htm")
        files = []
        if recursive:
            for dirpath, dirnames, filenames in os.walk(root_dir):
                # 出力先の reading フォルダ自身は対象から除外（無限に処理しないため）
                dirnames[:] = [d for d in dirnames if d != READING_DIRNAME]
                for fn in filenames:
                    if fn.lower().endswith(pattern_ext):
                        files.append(Path(dirpath) / fn)
        else:
            for fn in os.listdir(root_dir):
                p = root_dir / fn
                if p.is_file() and p.suffix.lower() in pattern_ext:
                    files.append(p)
        return sorted(files)

    def start_process(self):
        if self.is_running:
            return
        folder = self.folder_var.get().strip()
        if not folder:
            messagebox.showwarning("未指定", "対象フォルダを選択してください。")
            return
        root_dir = Path(folder)
        if not root_dir.is_dir():
            messagebox.showerror("エラー", "指定されたフォルダが見つかりません。")
            return

        self.is_running = True
        self.run_button.configure(state="disabled")
        self.log_box.configure(state="normal")
        self.log_box.delete("1.0", "end")
        self.log_box.configure(state="disabled")

        t = threading.Thread(target=self._process_worker, args=(root_dir,), daemon=True)
        t.start()

    def _process_worker(self, root_dir: Path):
        try:
            recursive = self.recursive_var.get()
            overwrite = self.overwrite_var.get()

            files = self.find_html_files(root_dir, recursive)
            total = len(files)
            self.log(f"対象HTMLファイル数: {total} 件")
            self.root.after(0, lambda: self.progress.configure(maximum=max(total, 1), value=0))

            if total == 0:
                self.log("処理対象のHTMLファイルが見つかりませんでした。")
                return

            success = 0
            skipped = 0
            failed = 0

            for i, src in enumerate(files, start=1):
                rel = src.relative_to(root_dir)
                dst = root_dir / READING_DIRNAME / rel

                if dst.exists() and not overwrite:
                    self.log(f"[スキップ] {rel} （既に出力先に存在）")
                    skipped += 1
                else:
                    try:
                        inject_tts_file(src, dst, log=self.log)
                        self.log(f"[完了] {rel} -> {READING_DIRNAME}/{rel}")
                        success += 1
                    except Exception as e:
                        failed += 1
                        self.log(f"[失敗] {rel}: {e}")
                        self.log(traceback.format_exc())

                self.root.after(0, lambda v=i: self.progress.configure(value=v))

            self.log("----")
            self.log(f"完了: {success} 件 / スキップ: {skipped} 件 / 失敗: {failed} 件")
            self.log(f"出力先: {root_dir / READING_DIRNAME}")

        except Exception as e:
            self.log(f"予期しないエラーが発生しました: {e}")
            self.log(traceback.format_exc())
        finally:
            self.is_running = False
            self.root.after(0, lambda: self.run_button.configure(state="normal"))


def main():
    root = tk.Tk()
    try:
        style = ttk.Style()
        if "vista" in style.theme_names():
            style.theme_use("vista")
        elif "clam" in style.theme_names():
            style.theme_use("clam")
    except Exception:
        pass
    app = TtsInjectorApp(root)
    root.mainloop()


if __name__ == "__main__":
    main()

# -*- coding: utf-8 -*-
"""
tts_core.py
HTMLファイルに「読み上げ機能」を注入するコアロジック。
- 本文テキストを文単位（。！？!?＋改行）で <span class="tts-sentence" id="tts-N"> に分割
- 再生中の文をハイライトし、自前計算によるスクロールで追従表示
- Web Speech API (speechSynthesis) を使うため、追加ライブラリ不要でブラウザだけで動作
"""

import re
import shutil
from pathlib import Path
from bs4 import BeautifulSoup, NavigableString, Comment

# 文の区切り（。！？!? の直後で分割。区切り文字自体は前の文に残す）
SENTENCE_SPLIT_RE = re.compile(r'(?<=[。！？!?])(?=\S|\s)')

SKIP_PARENT_TAGS = {'script', 'style', 'textarea', 'noscript', 'title'}

# common.js の中の「独自ページャー」を検出し、TTS連携用の
# window.__ttsReaderAPI を注入するためのアンカー文字列。
# このアンカーが見つかった場合のみパッチする（見つからなければ
# 元のファイルをそのままコピーし、TTS側は scrollIntoView にフォールバックする）。
COMMON_JS_PATCH_ANCHOR = "var btnNext = document.getElementById('btn-next');"

COMMON_JS_API_SNIPPET = """

\t/* ==== TTS reader integration (auto-injected by tts_core.py) ====
\t   centerElement() は、既存のページめくり機構（applyOffset / currentPage /
\t   ページ番号表示 / ボタンの有効無効状態）と矛盾なく、
\t   指定した要素を画面中央へ持ってくるための公開API。 */
\twindow.__ttsReaderAPI = {
\t\tcenterElement: function (el, animate) {
\t\t\tcomputeMetrics();
\t\t\tvar wrapperRect = wrapper.getBoundingClientRect();
\t\t\tvar elRect = el.getBoundingClientRect();
\t\t\tvar wrapperCenter = wrapperRect.left + wrapperRect.width / 2;
\t\t\tvar elCenter = elRect.left + elRect.width / 2;
\t\t\tvar delta = wrapperCenter - elCenter;
\t\t\tapplyOffset(scrollOffset + delta, animate !== false);
\t\t},
\t\t/* TTSバーの高さが変わった(画面回転・折り返し等)後に、
\t\t   ページ数やレイアウトを現在ページを保ったまま再計算する */
\t\trecalculate: function () {
\t\t\tcomputeMetrics();
\t\t\tgoToPage(Math.min(currentPage, totalPages - 1), false);
\t\t}
\t};
"""


def patch_common_js(text: str) -> str:
    """既知の #content-wrapper 型ページャー(common.js)にTTS連携APIを追加する。
    アンカーが見つからない場合は無加工のまま返す（別構造のサイトでも安全）。
    """
    if COMMON_JS_PATCH_ANCHOR in text and '__ttsReaderAPI' not in text:
        return text.replace(COMMON_JS_PATCH_ANCHOR, COMMON_JS_PATCH_ANCHOR + COMMON_JS_API_SNIPPET, 1)
    return text


_URL_SCHEME_RE = re.compile(r'^[a-zA-Z][a-zA-Z0-9+.\-]*:')


def is_local_relative_url(url: str) -> bool:
    """ローカルの相対パス参照（コピーして持ってくるべきアセット）かどうか判定する。
    http(s):, data:, mailto:, javascript:, # 始まり、 / 始まり(絶対パス) は対象外。
    """
    if not url:
        return False
    url = url.strip()
    if not url or url.startswith('#'):
        return False
    if url.startswith('//'):
        return False
    if url.startswith('/'):
        return False
    if _URL_SCHEME_RE.match(url):
        return False
    return True


def collect_asset_urls(soup: BeautifulSoup):
    """<link href>, <script src>, <img src> からローカル相対URLを収集する"""
    urls = []
    for tag_name, attr in (('link', 'href'), ('script', 'src'), ('img', 'src')):
        for el in soup.find_all(tag_name):
            u = el.get(attr)
            if u and is_local_relative_url(u):
                urls.append(u)
    return urls


def copy_local_assets(soup: BeautifulSoup, src_dir: Path, dst_dir: Path, log=None):
    """HTMLが相対参照しているcss/js/img等を、出力先にも同じ相対位置でコピーする。
    reading/ フォルダは元のHTMLより1階層深い場所に出力されるため、これをしないと
    css/js への相対リンクが壊れて見た目・動作が崩れる。
    common.js は既知の構造であればTTS連携APIを追加してコピーする。
    """
    for url in collect_asset_urls(soup):
        clean_url = url.split('?', 1)[0].split('#', 1)[0]
        if not clean_url:
            continue
        src_asset = (src_dir / clean_url).resolve()
        try:
            # src_dir の外を指すような不正な相対パス(../../etc)は安全のため無視
            src_dir_resolved = src_dir.resolve()
            src_asset.relative_to(src_dir_resolved.anchor)  # 単純な存在チェック用
        except Exception:
            pass
        if not src_asset.exists() or not src_asset.is_file():
            continue

        dst_asset = dst_dir / clean_url
        dst_asset.parent.mkdir(parents=True, exist_ok=True)

        if src_asset.name.lower() == 'common.js':
            try:
                text = src_asset.read_text(encoding='utf-8', errors='replace')
                patched = patch_common_js(text)
                dst_asset.write_text(patched, encoding='utf-8')
                if log and patched != text:
                    log(f"  (common.js に読み上げ連携APIを追加しました)")
            except Exception:
                shutil.copy2(src_asset, dst_asset)
        else:
            shutil.copy2(src_asset, dst_asset)

  function isElementVisible(el, margin) {

          // それでもダメな場合、もう一段階だけ再試行する

            requestAnimationFrame(function () {
              if (!isElementVisible(el, 4)) {
                doScrollIntoView(el);
              }
            });
          });
        }
      });
    });
  }

  }

  function loadVoices() {
    var all = window.speechSynthesis.getVoices() || [];
    voices = all.filter(function (v) { return v.lang && v.lang.toLowerCase().indexOf('ja') === 0; });
    if (voices.length === 0) voices = all; // 日本語音声が無ければ全部表示
    voiceSelect.innerHTML = '';
    voices.forEach(function (v, i) {
      var opt = document.createElement('option');
      opt.value = i;
      opt.textContent = v.name + ' (' + v.lang + ')';
      voiceSelect.appendChild(opt);
    });
    if (voices.length > 0) selectedVoice = voices[0];
  }

  if (typeof window.speechSynthesis !== 'undefined') {
    loadVoices();
    window.speechSynthesis.onvoiceschanged = loadVoices;
  }

  voiceSelect.addEventListener('change', function () {
    selectedVoice = voices[parseInt(voiceSelect.value, 10)];
  });

  rateInput.addEventListener('input', function () {
    rate = parseFloat(rateInput.value);
    rateLabel.textContent = rate.toFixed(1) + 'x';
  });

  // session: この呼び出しが発行された時点の playSession。
  // 発話の onend/onerror が実際に発火した時点で playSession が
  // 変わっていれば（＝途中でキャンセル/新規セッション開始済み）、
  // 古いイベントとして無視する。
  function speakFrom(i, session) {
    if (session !== playSession) return; // 既に無効化されたセッション
    if (i >= sentences.length) {
      isPlaying = false;
      idx = 0;
      clearHighlight();
      updateButtons();
      return;
    }
    idx = i;
    clearHighlight();
    var el = sentences[idx];
    el.classList.add('tts-active');
    scrollToElement(el);

    var utter = new SpeechSynthesisUtterance(el.textContent);
    utter.lang = (selectedVoice && selectedVoice.lang) || 'ja-JP';
    utter.rate = rate;
    if (selectedVoice) utter.voice = selectedVoice;
    utter.onend = function () {
      if (session !== playSession) return;
      if (isPlaying) speakFrom(idx + 1, session);
    };
    utter.onerror = function () {
      if (session !== playSession) return;
      if (isPlaying) speakFrom(idx + 1, session);
    };
    window.speechSynthesis.speak(utter);
  }

  // 新しい再生セッションを開始する（Play押下 / クリックでの位置指定 共通処理）
  function startPlaybackFrom(i) {
    playSession++; // 古いセッションのonend/onerrorを無効化
    var mySession = playSession;
    isPlaying = true;
    window.speechSynthesis.cancel();
    speakFrom(i, mySession);
    updateButtons();
  }

  btnPlay.addEventListener('click', function () {
    if (isPlaying) return;
    startPlaybackFrom(idx);
  });

  btnPause.addEventListener('click', function () {
    playSession++; // 以降届く古いonend/onerrorを無効化
    isPlaying = false;
    window.speechSynthesis.cancel();
    updateButtons();
  });

  btnStop.addEventListener('click', function () {
    playSession++;
    isPlaying = false;
    window.speechSynthesis.cancel();
    idx = 0;
    clearHighlight();
    updateButtons();
  });

  sentences.forEach(function (el, i) {
    el.style.cursor = 'pointer';
    el.addEventListener('click', function () {
      if (isPlaying) {
        startPlaybackFrom(i);
      } else {
        playSession++; // 念のため保留中のイベントを無効化
        window.speechSynthesis.cancel();
        idx = i;
        clearHighlight();
        el.classList.add('tts-active');
        scrollToElement(el);
      }
    });
  });

  updateButtons();
})();
"""


def split_sentences(text: str):
    parts = SENTENCE_SPLIT_RE.split(text)
    return [p for p in parts if p.strip() != '']


def inject_tts_soup(html: str) -> BeautifulSoup:
    """HTML文字列を受け取り、読み上げ機能を注入した BeautifulSoup を返す"""
    soup = BeautifulSoup(html, 'html5lib')

    body = soup.body
    if body is None:
        # body タグが無い断片的HTMLの場合はルートを body 扱いにする
        body = soup

    counter = [0]
    text_nodes = [t for t in body.find_all(string=True)]

    for text_node in text_nodes:
        if isinstance(text_node, Comment):
            continue
        parent = text_node.parent
        if parent is None or parent.name in SKIP_PARENT_TAGS:
            continue
        # 既に注入済みの span 内はスキップ（再実行対策）
        if parent.get('class') and 'tts-sentence' in parent.get('class'):
            continue
        text = str(text_node)
        if not text.strip():
            continue

        sentences = split_sentences(text)
        if not sentences:
            continue

        new_spans = []
        for s in sentences:
            counter[0] += 1
            span = soup.new_tag('span')
            span['class'] = 'tts-sentence'
            span['id'] = 'tts-%d' % counter[0]
            span.string = s
            new_spans.append(span)

        # text_node を new_spans の連続に置き換え
        text_node.replace_with(new_spans[0])
        prev = new_spans[0]
        for span in new_spans[1:]:
            prev.insert_after(span)
            prev = span

    # <link rel="stylesheet" href="css/tts-reader.css"> を head に追加
    link_tag = soup.new_tag('link', rel='stylesheet', href='css/' + TTS_CSS_FILENAME)
    if soup.head:
        soup.head.append(link_tag)
    else:
        if soup.html:
            soup.html.insert(0, link_tag)
        else:
            soup.insert(0, link_tag)

    # <script src="js/tts-reader.js"></script> を body 末尾に追加
    script_tag = soup.new_tag('script', src='js/' + TTS_JS_FILENAME)
    if soup.body:
        soup.body.append(script_tag)
    else:
        soup.append(script_tag)

    return soup


def inject_tts_html(html: str) -> str:
    """HTML文字列を受け取り、読み上げ機能を注入したHTML文字列を返す"""
    return str(inject_tts_soup(html))


def write_shared_tts_assets(dst_dir: Path, log=None):
    """全HTML共通の tts-reader.css / tts-reader.js を、指定ディレクトリ配下の
    css/ , js/ フォルダへ書き出す（既存の同名ファイルには常に最新内容で上書き）。
    """
    css_path = dst_dir / 'css' / TTS_CSS_FILENAME
    js_path = dst_dir / 'js' / TTS_JS_FILENAME
    css_path.parent.mkdir(parents=True, exist_ok=True)
    js_path.parent.mkdir(parents=True, exist_ok=True)
    css_path.write_text(TTS_CSS_CONTENT, encoding='utf-8')
    js_path.write_text(TTS_JS_CONTENT, encoding='utf-8')


def inject_tts_file(src_path: Path, dst_path: Path, encoding: str = 'utf-8', log=None):
    """1ファイルを読み込み→注入→書き出し。
    同時に、そのHTMLが相対参照しているcss/js/img等のローカルアセットも
    出力先の同じ相対位置へコピーする（reading/フォルダは1階層深くなるため、
    これをしないと相対リンクが壊れる）。
    読み上げ機能自体の共通css/jsは、各HTMLの隣にある css/ js/ フォルダへ
    tts-reader.css / tts-reader.js として書き出す（全HTML共通の1ファイル）。
    """
    with open(src_path, 'r', encoding=encoding, errors='replace') as f:
        html = f.read()

    soup = inject_tts_soup(html)

    dst_path.parent.mkdir(parents=True, exist_ok=True)
    with open(dst_path, 'w', encoding='utf-8') as f:
        f.write(str(soup))

    # アセットコピーは元のHTML基準で相対URLを解決するため、
    # 注入前のsoupではなく同じURL集合を再パースして取得する
    # （inject_tts_soup内で構造は変わるが href/src の値自体は変化しないため
    #   注入後のsoupからでも同じ結果になる）
    copy_local_assets(soup, src_path.parent, dst_path.parent, log=log)

    # 読み上げ機能自体の共通css/jsを書き出す（複数HTMLで同じフォルダを
    # 共有している場合は毎回同じ内容で上書きされるだけなので安全）
    write_shared_tts_assets(dst_path.parent, log=log)
