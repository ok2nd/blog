今回、色を付ける文字列を毎回指定する方法ではなく、Excelシートに対象文字列を列挙しておき、一括で処理するマクロを作成。（Claudeで作成。）

【 VBA（マクロ）作成 】

※ 前提：赤色＆ボールドにする文字列は、「ハイライト文字列」シートのA列に入っている。

Option Explicit

Sub 特定文字列を赤太字にする()

    Const LIST_SHEET As String = "ハイライト文字列"
    Const CMP_MODE As Long = vbBinaryCompare  ' 大文字小文字・全角半角を区別する。区別しないなら vbTextCompare

    Dim wsList As Worksheet, wsTarget As Worksheet
    Dim words() As String
    Dim lastRow As Long, i As Long, n As Long
    Dim rng As Range, c As Range
    Dim txt As String, pos As Long, w As Variant
    Dim hitCells As Long

    Set wsTarget = ActiveSheet

    On Error Resume Next
    Set wsList = ThisWorkbook.Worksheets(LIST_SHEET)
    If wsList Is Nothing Then Set wsList = wsTarget.Parent.Worksheets(LIST_SHEET)
    On Error GoTo 0

    If wsList Is Nothing Then
        MsgBox "「" & LIST_SHEET & "」シートが見つかりません。", vbExclamation
        Exit Sub
    End If

    If wsTarget.Name = LIST_SHEET Then
        MsgBox "ハイライト文字列シート以外のシートを表示してから実行してください。", vbExclamation
        Exit Sub
    End If

    ' --- A列の文字列を配列に取り込む（空白は除く） ---
    lastRow = wsList.Cells(wsList.Rows.Count, "A").End(xlUp).Row
    ReDim words(1 To Application.Max(lastRow, 1))
    For i = 1 To lastRow
        If Len(Trim$(CStr(wsList.Cells(i, "A").Value))) > 0 Then
            n = n + 1
            words(n) = CStr(wsList.Cells(i, "A").Value)
        End If
    Next i

    If n = 0 Then
        MsgBox "ハイライト文字列シートのA列に文字列がありません。", vbExclamation
        Exit Sub
    End If

    ' --- 対象セル（定数の文字列セルのみ。数式セルは部分書式不可のため除外） ---
    On Error Resume Next
    Set rng = wsTarget.UsedRange.SpecialCells(xlCellTypeConstants, xlTextValues)
    On Error GoTo 0

    If rng Is Nothing Then
        MsgBox "処理対象の文字列セルがありません。", vbInformation
        Exit Sub
    End If

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual

    On Error Resume Next
    For Each c In rng.Cells
        txt = CStr(c.Value)
        Dim hit As Boolean: hit = False
        For i = 1 To n
            pos = InStr(1, txt, words(i), CMP_MODE)
            Do While pos > 0
                With c.Characters(pos, Len(words(i))).Font
                    .Color = vbRed
                    .Bold = True
                End With
                hit = True
                pos = InStr(pos + Len(words(i)), txt, words(i), CMP_MODE)
            Loop
        Next i
        If hit Then hitCells = hitCells + 1
    Next c
    On Error GoTo 0

    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True

    MsgBox hitCells & " 個のセルを処理しました。", vbInformation

End Sub

【 VBA（マクロ）使い方 】

  1. Alt + F11 で VBE を開く。
  2. 挿入 → 標準モジュール を選び、上のコードを貼り付ける。
  3. 処理したいシートを表示した状態で、Alt + F8 → 特定文字列を赤太字にする → 実行。
