■《JavaScript修正》

1. マップの初期表示を「日本全土」に設定する

【修正前】

var map = L.map('map');

【修正後】

// 日本のほぼ中心を指定し、ズームレベル5（日本全体が見える程度）で初期化
var map = L.map('map').setView([36.0, 138.0], 5);

2. タイルレイヤーを即座に表示する

【修正前】

var tileLayer = L.tileLayer('https://mt1.google.com/vt/lyrs=r&x={x}&y={y}&z={z}', {
    maxZoom: 18,
    attribution: 'Google Map'
});

【修正後】

var tileLayer = L.tileLayer('https://mt1.google.com/vt/lyrs=r&x={x}&y={y}&z={z}', {
    maxZoom: 18,
    attribution: 'Google Map'
});
tileLayer.addTo(map); // ← 追加

3. 自動的な位置取得（locate）の設定変更

【修正前】

map.locate({setView: true, maxZoom: 16, timeout: 20000});

【修正後】

// setViewをfalseにし、位置が取れた時（onLocationFound）に移動させる
map.locate({setView: false, timeout: 20000});

4. onLocationFound 関数内でズームレベルを調整

【修正前】

function onLocationFound(e) {
    var centerIcon = L.divIcon({ className: 'center-icon', iconAnchor: [8, 8] });
    L.marker(e.latlng, {icon: centerIcon}).addTo(map);
}

【修正後】

function onLocationFound(e) {
    var centerIcon = L.divIcon({ className: 'center-icon', iconAnchor: [8, 8] });
    L.marker(e.latlng, {icon: centerIcon}).addTo(map);
    // 現在地にスムーズに移動（ズームレベル16）
    map.setView(e.latlng, 16); 
}
