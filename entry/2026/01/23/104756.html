PHP）MySQL関数新旧「mysqli_」vs「mysql_」比較表。

2026-01-23 · ホームページ、Web作成 コンピュータ

PHPのMySQL関数には「mysql_」で始まる関数と「mysqli_」で始まる関数の2種類ある。

「mysql_」は古いタイプで、PHP 5.5で非推奨、7.0で削除された。

「mysqli_」は、セキュリティ強化、オブジェクト指向対応、トランザクション対応。

MySQL関数の新旧比較表を作成。「mysqli_」型には、手続き型とオブジェクト型があるが、手続き型との比較一覧表を作成。

実際にコーディングする例で記述。

カテゴリ mysql_（旧式） mysqli_（手続き型）
接続・選択 mysql_connect($host, $user, $pass) mysqli_connect($host, $user, $pass, $db)
  mysql_pconnect($host, $user, $pass) mysqli_connect('p:'.$host, $user, $pass)
  mysql_select_db($db, $conn) mysqli_select_db($conn, $db)
クエリ実行 mysql_query($sql, $conn) mysqli_query($conn, $sql)
  mysql_unbuffered_query($sql, $conn) mysqli_query($conn, $sql, MYSQLI_USE_RESULT)
  mysql_db_query($db, $sql, $conn) mysqli_select_db($conn, $db); mysqli_query($conn, $sql)
データ取得 mysql_fetch_assoc($res) mysqli_fetch_assoc($res)
  mysql_fetch_array($res, $type) mysqli_fetch_array($res, $type)
  mysql_fetch_row($res) mysqli_fetch_row($res)
  mysql_fetch_object($res, $class) mysqli_fetch_object($res, $class)
  mysql_result($res, $row, $field) (なし) ※mysqli_data_seek後にfetchで代用
情報取得 mysql_num_rows($res) mysqli_num_rows($res)
  mysql_num_fields($res) mysqli_num_fields($res)
  mysql_affected_rows($conn) mysqli_affected_rows($conn)
  mysql_insert_id($conn) mysqli_insert_id($conn)
  mysql_data_seek($res, $row) mysqli_data_seek($res, $row)
フィールド情報 mysql_fetch_field($res, $off) mysqli_fetch_field($res)
  mysql_field_name($res, $off) mysqli_fetch_field_direct($res, $off)->name
  mysql_field_type($res, $off) mysqli_fetch_field_direct($res, $off)->type
  mysql_field_len($res, $off) mysqli_fetch_field_direct($res, $off)->length
  mysql_field_table($res, $off) mysqli_fetch_field_direct($res, $off)->table
  mysql_field_flags($res, $off) mysqli_fetch_field_direct($res, $off)->flags
エラー・状態 mysql_error($conn) mysqli_error($conn)
  mysql_errno($conn) mysqli_errno($conn)
  mysql_stat($conn) mysqli_stat($conn)
  mysql_thread_id($conn) mysqli_thread_id($conn)
  mysql_client_encoding($conn) mysqli_character_set_name($conn)
  mysql_get_client_info() mysqli_get_client_info($conn)
  mysql_get_host_info($conn) mysqli_get_host_info($conn)
  mysql_get_proto_info($conn) mysqli_get_proto_info($conn)
  mysql_get_server_info($conn) mysqli_get_server_info($conn)
運用・管理 mysql_real_escape_string($s, $c) mysqli_real_escape_string($c, $s)
  mysql_set_charset($cs, $conn) mysqli_set_charset($conn, $cs)
  mysql_ping($conn) mysqli_ping($conn)
  mysql_list_dbs($conn) mysqli_query($conn, SHOW DATABASES)
  mysql_list_tables($db, $conn) mysqli_query($conn, SHOW TABLES FROM $db)
  mysql_list_fields($db, $tbl, $c) mysqli_query($c, SHOW COLUMNS FROM $tbl)
  mysql_free_result($res) mysqli_free_result($res)
  mysql_close($conn) mysqli_close($conn)
