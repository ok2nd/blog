画像フォーマットを変換するWindowsアプリ作成。複数ファイル一括も可。

変換できるフォーマット形式は以下。

  • インプット形式：JPEG、PNG、WebP、AVIF、GIF、SVG、TIFF、BMP、HEIC/HEIF
  • アウトプット形式：JPEG、PNG、WebP、AVIF
■ Pythonアプリ
  • 変換したい画像ファイル（複数も可）をドラッグ＆ドロップでウインドウに入れる。
  • サムネイルで画像一覧が表示される。画像を選択すると、現在の情報が表示される。
  • 出力フォーマットを選択する。JPEG、WebP、AVIFは品質レベルを選択できる。初期値は、そのフォーマットで一般的に推奨される品質レベル。AVIFは品質レベルを上げてもファイルサイズがあまり大きくならないので、80にしても良いかも。
    • JPEG：70～95（初期値：85）
    • PNG
    • WebP：65～90（初期値：80）
    • AVIF：65～90（初期値：70）
  • 「フォーマット変換」ボタンで変換され、それぞれのフォーマット名のフォルダーに格納される。品質レベルもフォルダー名に付与される。

■ フォーマット別ファイルサイズ比較

今回作成したアプリでフォーマット変換すると、WebPとAVIFは、元のJPEGと比較してかなり小さくなる。画質の違いは、よく分からない。

しかし、これらの変換後ファイルを、はてなブログにアップロードすると、サイズが変わる。JPEGは変わらないが、PNGは若干増える。WebPは何故か6倍ほど大きくなる。AVIFは半分近くに小さくなる。はてなブログにアップロードするには、WebPは向いていない。

#  ImageEntry  – holds one input file
# ═══════════════════════════════════════════════════════════════════════════════

class ImageEntry:
    def __init__(self, path: Path):
        self.path    = path
        self.name    = path.name
        self.ext     = path.suffix.lower()
        self.format  = EXT_TO_FORMAT.get(self.ext, "Unknown")
        self._thumb: ImageTk.PhotoImage | None = None
        self._dims:  tuple[int, int] | None    = None

    # ── thumbnail ──────────────────────────────────────────────────────────
    def get_thumbnail(self) -> "ImageTk.PhotoImage | None":
        if self._thumb is not None:
            return self._thumb
        try:
            img = self._open()
            if img is None:
                return None
            if not self._dims:
                self._dims = img.size
            img.thumbnail((THUMB_W, THUMB_H), Image.LANCZOS)
            self._thumb = ImageTk.PhotoImage(img)
            return self._thumb
        except Exception:
            return None

    # ── dimensions ────────────────────────────────────────────────────────
    def get_dims(self) -> "tuple[int,int] | None":
        if self._dims:
            return self._dims
        try:
            img = self._open()
            if img:
                self._dims = img.size
        except Exception:
            pass
        return self._dims

    # ── file size string ───────────────────────────────────────────────────
    def file_size_str(self) -> str:
        try:
            n = self.path.stat().st_size
            if n >= 1_048_576:
                return f"{n/1_048_576:.1f} MB"
            elif n >= 1_024:
                return f"{n/1_024:.1f} KB"
            return f"{n} B"
        except Exception:
            return "—"

    # ── image opening ──────────────────────────────────────────────────────
    def _open(self) -> "Image.Image | None":
        if self.ext == ".svg":
            return _open_svg(self.path)
        img = Image.open(self.path)
        img.load()
        return img


def _open_svg(path: Path) -> "Image.Image | None":
    if HAS_CAIROSVG:
        data = cairosvg.svg2png(url=str(path), output_width=800)
        return Image.open(io.BytesIO(data))
    if HAS_SVGLIB:
        drawing = svg2rlg(str(path))
        if drawing:
            data = renderPM.drawToString(drawing, fmt="PNG")
            return Image.open(io.BytesIO(data))
    return None


# ═══════════════════════════════════════════════════════════════════════════════
#  ThumbnailCard  – a single thumbnail tile
# ═══════════════════════════════════════════════════════════════════════════════

class ThumbnailCard(tk.Frame):
    def __init__(self, parent, entry: ImageEntry, index: int, on_select, **kw):
        super().__init__(parent, bg=C["surface"], cursor="hand2",
                         highlightthickness=2,
                         highlightbackground=C["border"],
                         width=CELL_W, height=CELL_H, **kw)
        self.pack_propagate(False)
        self.entry     = entry
        self.index     = index
        self.on_select = on_select
        self.selected  = False
        self._build()
        self._setup_bindings()

    def _build(self):
        # ── image area ───────────────────────────────────────────────────
        img_frame = tk.Frame(self, bg=C["surface2"], width=THUMB_W+8, height=THUMB_H+6)
        img_frame.pack(pady=(8, 3))
        img_frame.pack_propagate(False)

        thumb = self.entry.get_thumbnail()
        if thumb:
            self.img_lbl = tk.Label(img_frame, image=thumb, bg=C["surface2"])
            self.img_lbl.image = thumb
        else:
            self.img_lbl = tk.Label(img_frame, text="?",
                                    font=("Segoe UI", 20), fg=C["placeholder"],
                                    bg=C["surface2"])
        self.img_lbl.place(relx=0.5, rely=0.5, anchor="center")

        # ── filename ─────────────────────────────────────────────────────
        name = self.entry.name
        if len(name) > 17:
            name = name[:14] + "…"
        self.name_lbl = tk.Label(self, text=name, bg=C["surface"],
                                 fg=C["subtext"], font=("Segoe UI", 8),
                                 wraplength=CELL_W - 8)
        self.name_lbl.pack()

        # ── format badge ─────────────────────────────────────────────────
        self.badge = tk.Label(self, text=self.entry.format,
                              bg=C["accent_bg"], fg=C["accent"],
                              font=("Segoe UI", 7, "bold"),
                              padx=4, pady=1)
        self.badge.pack(pady=(1, 0))

        self._all = [self, img_frame, self.img_lbl, self.name_lbl, self.badge]

    def _setup_bindings(self):
        for w in self._all:
            w.bind("<Button-1>", lambda e: self.on_select(self.index))
            w.bind("<Enter>",    lambda e: self._hover(True))
            w.bind("<Leave>",    lambda e: self._hover(False))

    def set_selected(self, sel: bool):
        self.selected = sel
        bg = C["thumb_sel"] if sel else C["surface"]
        hl = C["accent"]    if sel else C["border"]
        self.config(bg=bg, highlightbackground=hl,
                    highlightthickness=2 if sel else 2)
        self.name_lbl.config(bg=bg)
        self.badge.config(bg=C["accent_bg"] if not sel else C["accent"],
                          fg=C["accent"]    if not sel else "#FFFFFF")

    def _hover(self, on: bool):
        if self.selected:
            return
        bg = C["thumb_hover"] if on else C["surface"]
        self.config(bg=bg)
        self.name_lbl.config(bg=bg)

        self._status = tk.StringVar(value="ファイルをドラッグ＆ドロップしてください")
        tk.Label(sb, textvariable=self._status, bg=C["status_bg"],
                 fg=C["subtext"], font=("Segoe UI", 9),
                 anchor="w").pack(fill=tk.BOTH, padx=12, pady=5)

    # ═══════════════════════════════════════════════════════════════════════
    #  Drag & Drop
    # ═══════════════════════════════════════════════════════════════════════

    def _setup_dnd(self):
        if not HAS_DND:
            return
        for w in [self.root, self._canvas, self._drop_hint, self._thumb_host]:
            try:
                w.drop_target_register(DND_FILES)
                w.dnd_bind("<<Drop>>", self._on_drop)
            except Exception:
                pass

    def _on_drop(self, event):
        raw = self.root.tk.splitlist(event.data)
        self._add_files([Path(p) for p in raw])

    # ═══════════════════════════════════════════════════════════════════════
    #  File management
    # ═══════════════════════════════════════════════════════════════════════

    def _add_files(self, paths: list[Path]):
        added = 0
        skipped_fmt = 0
        for p in paths:
            if not p.is_file():
                continue
            if p.suffix.lower() not in SUPPORTED_EXT:
                skipped_fmt += 1
                continue
            if not any(e.path == p for e in self.entries):
                self.entries.append(ImageEntry(p))
                added += 1

        if added == 0:
            if skipped_fmt:
                self._status.set(f"サポート外のファイル形式です ({skipped_fmt}件)")
            return

        self.entries.sort(key=lambda e: e.name.lower())
        self._refresh_grid()

        if self.selected_idx < 0:
            self._select(0)

        self._status.set(f"{len(self.entries)} 件のファイルを読み込みました")

    def _clear_all(self):
        self.entries.clear()
        self.selected_idx = -1
        self._refresh_grid()
        self._info_name.config(text="(未選択)")
        self._info_fmt.config(text="—")
        self._info_sz.config(text="—")
        self._info_dim.config(text="—")
        self._status.set("ファイルをドラッグ＆ドロップしてください")

    # ═══════════════════════════════════════════════════════════════════════
    #  Thumbnail grid
    # ═══════════════════════════════════════════════════════════════════════

    def _refresh_grid(self):
        for c in self.cards:
            c.destroy()
        self.cards.clear()

        n = len(self.entries)
        self._count_lbl.config(text=f"ファイル一覧  ({n}件)")

        if n == 0:
            self._show_hint(True)
            return

        self._show_hint(False)

        for i, entry in enumerate(self.entries):
            card = ThumbnailCard(self._thumb_host, entry, i,
                                 on_select=self._select)
            card.grid(row=i // THUMB_COLS, column=i % THUMB_COLS,
                      padx=5, pady=5)
            self.cards.append(card)

        for i, card in enumerate(self.cards):
            card.set_selected(i == self.selected_idx)

        self._canvas.update_idletasks()
        self._canvas.configure(scrollregion=self._canvas.bbox("all"))

    def _show_hint(self, show: bool):
        if show:
            self._canvas.pack_forget()
            self._vbar.pack_forget()
            self._drop_hint.place(relx=0, rely=0, relwidth=1, relheight=1)
        else:
            self._drop_hint.place_forget()
            self._vbar.pack(side=tk.RIGHT, fill=tk.Y)
            self._canvas.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)

    def _select(self, index: int):
        if index < 0 or index >= len(self.entries):
            return
        old = self.selected_idx
        self.selected_idx = index

        if 0 <= old < len(self.cards):
            self.cards[old].set_selected(False)
        if 0 <= index < len(self.cards):
            self.cards[index].set_selected(True)

        e = self.entries[index]
        self._info_name.config(text=e.name)
        self._info_fmt.config(text=f"フォーマット:  {e.format}")
        self._info_sz.config(text=f"ファイルサイズ:  {e.file_size_str()}")
        dims = e.get_dims()
        self._info_dim.config(
            text=f"解像度:  {dims[0]} × {dims[1]} px" if dims else "解像度:  —")

    # ═══════════════════════════════════════════════════════════════════════
    #  Format / Quality controls
    # ═══════════════════════════════════════════════════════════════════════

    def _on_format_change(self, *_):
        fmt = self.out_format.get()
        if fmt in QUALITY_CONFIG:
            cfg = QUALITY_CONFIG[fmt]
            self._q_scale.config(from_=cfg["min"], to=cfg["max"])
            q = cfg["default"]
            self.quality_var.set(q)
            self._q_val.config(text=str(q))
            self._q_min.config(text=str(cfg["min"]))
            self._q_max.config(text=str(cfg["max"]))
            self._q_outer.pack(fill=tk.X, pady=(0, 10))
        else:
            self._q_outer.pack_forget()

    def _on_quality_drag(self, raw):
        # snap to multiples of 5
        v = round(float(raw) / 5) * 5
        fmt = self.out_format.get()
        if fmt in QUALITY_CONFIG:
            cfg = QUALITY_CONFIG[fmt]
            v = max(cfg["min"], min(cfg["max"], v))
        if self.quality_var.get() != v:
            self.quality_var.set(v)
        self._q_val.config(text=str(v))

    # ═══════════════════════════════════════════════════════════════════════
    #  Conversion
    # ═══════════════════════════════════════════════════════════════════════

    def _convert_all(self):
        if self._converting:
            return
        if not self.entries:
            messagebox.showwarning("警告", "変換するファイルがありません。")
            return

        out_fmt = self.out_format.get()
        quality = self.quality_var.get()
        entries = list(self.entries)

        self._converting = True
        self._cvt_btn.config(state=tk.DISABLED, text="変換中…")
        self._prog_bar.config(maximum=len(entries), value=0)
        self._prog_bar.pack(fill=tk.X, pady=(4, 2))
        self._prog_lbl.pack(fill=tk.X)

        threading.Thread(
            target=self._worker,
            args=(out_fmt, quality, entries),
            daemon=True,
        ).start()

    def _worker(self, out_fmt: str, quality: int, entries: list[ImageEntry]):
        ok, errors = 0, []
        ext = FORMAT_EXT[out_fmt]

        for i, entry in enumerate(entries):
            self.root.after(0, lambda n=entry.name:
                            self._status.set(f"変換中:  {n}"))
            try:
                if out_fmt in QUALITY_CONFIG:
                    folder_name = f"{out_fmt}-{quality}"
                else:
                    folder_name = out_fmt
                out_dir = entry.path.parent / folder_name
                out_dir.mkdir(parents=True, exist_ok=True)
                out_path = out_dir / (entry.path.stem + ext)

                img = self._load(entry)
                if img is None:
                    errors.append(f"{entry.name}:  SVG 読み込み失敗")
                    continue
                self._save(img, out_path, out_fmt, quality)
                ok += 1
            except Exception as exc:
                errors.append(f"{entry.name}:  {exc}")

            p = i + 1
            self.root.after(0, lambda v=p, t=len(entries):
                            self._set_progress(v, t))

        self.root.after(0, lambda: self._done(ok, errors))

    def _load(self, entry: ImageEntry) -> "Image.Image | None":
        if entry.ext == ".svg":
            return _open_svg(entry.path)
        img = Image.open(entry.path)
        # handle animated → take first frame
        try:
            img.seek(0)
        except Exception:
            pass
        img.load()
        return img

    def _save(self, img: Image.Image, path: Path, fmt: str, quality: int):
        # colour mode coercion
        if fmt == "JPEG":
            img = _to_rgb(img)
        elif fmt in ("WebP", "AVIF", "PNG"):
            if img.mode == "P":
                img = img.convert("RGBA")
            elif img.mode not in ("RGB", "RGBA", "L", "LA"):
                img = img.convert("RGBA")

        kw: dict = {}
        if fmt == "JPEG":
            kw = {"quality": quality, "optimize": True}
        elif fmt == "WebP":
            kw = {"quality": quality, "method": 6}
        elif fmt == "AVIF":
            kw = {"quality": quality}
        # PNG → no quality arg

        img.save(str(path), format=fmt, **kw)

    # ── progress / done ────────────────────────────────────────────────────

    def _set_progress(self, v: int, total: int):
        self._prog_bar["value"] = v
        self._prog_lbl.config(text=f"{v} / {total} 件完了")

    def _done(self, ok: int, errors: list[str]):
        self._converting = False
        self._cvt_btn.config(state=tk.NORMAL, text="フォーマット変換")
        self._prog_bar.pack_forget()
        self._prog_lbl.pack_forget()

        fmt = self.out_format.get()
        quality = self.quality_var.get()
        folder_name = f"{fmt}-{quality}" if fmt in QUALITY_CONFIG else fmt
        if errors:
            msg = f"{ok} 件の変換に成功しました。\n\n"
            msg += "エラー:\n" + "\n".join(errors[:12])
            if len(errors) > 12:
                msg += f"\n…他 {len(errors)-12} 件"
            messagebox.showerror("変換完了（エラーあり）", msg)
        else:
            messagebox.showinfo(
                "変換完了",
                f"{ok} 件すべての変換が完了しました！\n\n"
                f"出力先:  [元ファイルのフォルダー] /{folder_name}/ フォルダー内")

        self._status.set(
            f"変換完了:  {ok} 件成功" +
            (f"  /  {len(errors)} 件エラー" if errors else ""))


# ═══════════════════════════════════════════════════════════════════════════════
#  Helpers
# ═══════════════════════════════════════════════════════════════════════════════

def _to_rgb(img: Image.Image) -> Image.Image:
    """Convert any mode to RGB, compositing alpha over white."""
    if img.mode == "RGB":
        return img
    if img.mode == "P":
        img = img.convert("RGBA")
    if img.mode in ("RGBA", "LA"):
        bg = Image.new("RGB", img.size, (255, 255, 255))
        alpha = img.split()[-1]
        bg.paste(img.convert("RGB"), mask=alpha)
        return bg
    return img.convert("RGB")
