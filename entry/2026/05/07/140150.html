JavaScriptでRSSを取得できない（CORS制限）を Cloudflare Workers で無料解決する方法。PHP不要。はてなブックマークRSS取得。

2026-05-07 · ホームページ、Web作成 はてなブログ

自分用のサイトで、はてなブックマークの最新情報をRSSリーダーで表示している。（以下のどちらか）

  • https://feeds.feedburner.com/hatena/b/hotentry/
  • https://b.hatena.ne.jp/hotentry.rss

はてなブックマークのRSSを取得するのに、CORS（Cross-Origin Resource Sharing）制限を回避するために使用していた「api.allorigins.win」のプロキシサービスがあまり調子よくない（？）ので、「Cloudflare Workers」を使ってみる。

「Cloudflare Workers」とは、サーバーを持たずにコードをインターネット上で実行できるサービス。CORS制限で、ブラウザから直接外部RSSを取得できない場合に、「Cloudflare Workers」が使える。無料ホスティング（InfinityFreeなど）では fetch によるプログラム的なリクエストが遮断されることがあり、PHPによるプロキシも動作しない場合がある。Cloudflare Workers はその制約を受けないため、安定して動作する。

《 無料枠の範囲 》

  • 1日10万リクエストまで無料
  • 世界中のエッジで動くため高速
  • 独自ドメイン不要（*.workers.dev のURLが自動発行）
【CORS回避RSSプロキシを使う手順】

「https://workers.cloudflare.com」にアクセスしてアカウント作成。

以下の手順で、Cloudflare Workersを作成する。

１．Build ＞ Compute

２．Build ＞ Compute ＞ Workers & Pages

「Create application」をクリック。

３．「Start with Hello World!」を選択

４．「Worker name」を確認して「Deploy」

５．Edit code

６．以下のコードを貼り付ける。

export default {
  async fetch(request) {
    // CORS プリフライト対応
    if (request.method === 'OPTIONS') {
      return new Response(null, {
        status: 204,
        headers: corsHeaders(),
      });
    }

    const { searchParams } = new URL(request.url);
    const targetUrl = searchParams.get('url');

    if (!targetUrl) {
      return jsonResponse({ contents: null, status: { error: 'url parameter is required' } }, 400);
    }

    try {
      const res = await fetch(targetUrl, {
        headers: { 'User-Agent': 'Mozilla/5.0 (compatible; CFProxy/1.0)' },
      });

      const body = await res.text();

      return jsonResponse({
        contents: body,
        status: {
          url: res.url,
          content_type: res.headers.get('content-type'),
          http_code: res.status,
        },
      });
    } catch (e) {
      return jsonResponse({ contents: null, status: { error: e.message } }, 502);
    }
  },
};

function corsHeaders() {
  return {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Methods': 'GET, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type',
  };
}

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      'Content-Type': 'application/json; charset=utf-8',
      ...corsHeaders(),
    },
  });
}

《 RSSリーダーを使うJavaScript 》

以下のようなスクリプトで「Cloudflare Workers」を使って、RSSを取得する。

<script>
// const rssUrl = 'https://b.hatena.ne.jp/hotentry.rss';
const rssUrl = 'https://feeds.feedburner.com/hatena/b/hotentry';
async function fetchRSS() {
	const response = await fetch(`https://（Worker name）.workers.dev/?url=${encodeURIComponent(rssUrl)}`);
	const data = await response.json();
	const parser = new DOMParser();
	const xmlDoc = parser.parseFromString(data.contents, "text/xml");
	// RSSフィードをパース
	const items = xmlDoc.querySelectorAll("item");
	let feeds = [];
	items.forEach(item => {
		const title = item.querySelector("title")?.textContent || "No Title";
		const link = item.querySelector("link")?.textContent || "#";
		// pubDate の取得を試みるが、はてなブックマークのRSSでは
		// 日付が dc:date 形式のため、通常の querySelector では取得できない場合がある。
		let pubDate = item.querySelector("pubDate")?.textContent || item.querySelector("dc\\:date")?.textContent || "";
		feeds.push({ title, link, pubDate });
	});
	return feeds;
}
function displayFeeds(feeds) {
	const feedsContainer = document.getElementById('rss-feeds');
	feeds.forEach(feed => {
		const feedElement = document.createElement('ul');
		feedElement.className = 'feed';
		feedElement.innerHTML = `
			<li><a href="${feed.link}" target="_blank">${feed.title}</a></li>
		`;
		feedsContainer.appendChild(feedElement);
	});
}
const params = new URLSearchParams(window.location.search);
if (params.get("rss") !== "n") {
	document.addEventListener('DOMContentLoaded', async () => {
		const feeds = await fetchRSS();
		document.getElementById("feeds-title").style.display = "block";
		document.getElementById("rss-feeds").style.display = "block";
		displayFeeds(feeds);
	});
}
</script>
