≪HTMLとCSS≫

<style>
.line {
	position: relative;
	height: 1.5em;
}
.line::before {
	content: "";
	display: block;
	width: 100%;
	position: absolute;
	margin-top: 0.5em;
	border-top: 1px dashed silver;
}
.line .name {
	position: absolute;
	padding-right: 1ex;
	background-color: white;
}
.line .page {
	right: 0;
	position: absolute;
	padding-left: 1ex;
	background-color: white;
}
</style>
